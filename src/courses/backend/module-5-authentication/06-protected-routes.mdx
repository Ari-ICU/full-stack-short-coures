## 🚦 ការការពារ API Routes

ឥឡូវនេះ យើងមាន `requireAuth` និង `requireRole` រួចរាល់ហើយ។ តោះយើងយកវាទៅប្រើប្រាស់ក្នុង Router ជាក់ស្តែង។

```tabs
javascript
// routes/adminRoutes.js
import express from 'express';
import { requireAuth, requireRole } from '../middlewares/authMiddleware.js';

const router = express.Router();

// Route ធម្មតា: នរណាក៏អាចចូលមើលបាន
router.get('/public', (req, res) => {
    res.json({ message: "ទិន្នន័យសាធារណៈ" });
});

// Route ការពារទី១: ត្រូវតែមានគណនី (Login រួច) ទើបចូលមើលបាន
router.get('/profile', requireAuth, (req, res) => {
    res.json({ 
        message: "ព័ត៌មានផ្ទាល់ខ្លួនរបស់អ្នក",
        userId: req.user.id 
    });
});

// Route ការពារទី២: ត្រូវតែជា Admin ទើបអាចលុបទិន្នន័យបាន!
// យើងអាចដាក់ Middleware ចូលគ្នាបានច្រើនតាមចិត្ត
router.delete('/users/:id', requireAuth, requireRole('Admin'), (req, res) => {
    res.json({ message: `បានលុប User ${req.params.id} ដោយជោគជ័យ` });
});

// Route ការពារទី៣: Manager ក៏បាន Admin ក៏បាន
router.get('/reports', requireAuth, requireRole('Admin', 'Manager'), (req, res) => {
    res.json({ message: "ទិន្នន័យរបាយការណ៍ហិរញ្ញវត្ថុ" });
});

export default router;
```

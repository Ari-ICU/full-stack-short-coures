## 🔒 គ្រោះថ្នាក់នៃការផ្ទុកលេខសម្ងាត់ផ្ទាល់ (Plain-text Passwords)

ប្រសិនបើអ្នកបង្កើតមុខងារចុះឈ្មោះ (Register) ដោយយកលេខសម្ងាត់ដែលអ្នកប្រើប្រាស់វាយបញ្ចូល ទៅរក្សាទុកក្នុង Database ផ្ទាល់ នេះគឺជា**កំហុសដ៏ធំបំផុតនៅក្នុង Backend**។

ប្រសិនបើ Hacker អាចលួចចូល Database របស់អ្នកបាន ពួកគេនឹងបានលេខសម្ងាត់ពិតប្រាកដរបស់អ្នកប្រើប្រាស់ទាំងអស់ ដែលអាចអោយពួកគេយកទៅសាកល្បង Login គណនីផ្សេងៗ (ដូចជា Facebook, ធនាគារ) របស់អ្នកប្រើប្រាស់ទាំងនោះបាន ព្រោះមនុស្សភាគច្រើនប្រើប្រាស់លេខសម្ងាត់តែមួយគ្រប់កន្លែង។

### Hashing ជាមួយ bcrypt

**Hashing** គឺជាការយកទិន្នន័យទៅក្រឡុកតាមរូបមន្តគណិតវិទ្យា ទៅជាទម្រង់អក្សរញ៉េរញ៉ៃ ដែល**មិនអាច**បកប្រែត្រលប់មកវិញបានឡើយ ទោះបីជាអ្នកបង្កើតរូបមន្តនោះក៏ដោយ (One-way function)។

Library ដែលពេញនិយមបំផុតគឺ `bcrypt`។

```bash
npm install bcrypt
```

```tabs
javascript
// authService.js
import bcrypt from 'bcrypt';

// 1. ការបង្កើត Hashed Password (ពេល Register)
export const hashPassword = async (plainPassword) => {
    const saltRounds = 10; // ចំនួនដងនៃការក្រឡុកអោយកាន់តែស្មុគស្មាញ (ចំណាយពេលប្រហែល 0.1 វិនាទី)
    const hashedPassword = await bcrypt.hash(plainPassword, saltRounds);
    return hashedPassword; 
    // លទ្ធផល: $2b$10$X8... (រក្សាទុកអាហ្នឹងក្នុង Database)
};

// 2. ការផ្ទៀងផ្ទាត់ (ពេល Login)
export const comparePassword = async (plainPassword, hashedPasswordFromDB) => {
    // bcrypt នឹងយក Password ថ្មីទៅក្រឡុក ហើយប្រៀបធៀបជាមួយ Hashed ចាស់
    const isMatch = await bcrypt.compare(plainPassword, hashedPasswordFromDB);
    return isMatch; // true or false
};
```

> [!CAUTION]
> កុំប្រើប្រាស់ `md5` ឫ `sha1` សម្រាប់ Hash លេខសម្ងាត់អោយសោះ ព្រោះកុំព្យូទ័រជំនាន់ថ្មីអាចទាយ (Crack) វារកលេខសម្ងាត់ដើមវិញបានក្នុងរយៈពេលត្រឹមតែប៉ុន្មានវិនាទីប៉ុណ្ណោះ។ ត្រូវប្រើ `bcrypt` ឫ `argon2` ជានិច្ច!

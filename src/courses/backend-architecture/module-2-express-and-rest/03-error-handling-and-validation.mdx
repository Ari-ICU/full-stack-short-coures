## 🛑 ការគ្រប់គ្រងកំហុស (Error Handling) ក្នុង Express

ជាធម្មតានៅពេលកូដមានបញ្ហា (ឧ. អថេរគ្មានតម្លៃ ឬដាច់ Internet ពេលភ្ជាប់ Database) Server របស់អ្នកអាចនឹងគាំង (Crash) ហើយឈប់ដំណើរការតែម្តង។ ដើម្បីការពារបញ្ហានេះ Express មានយន្តការមួយហៅថា **Global Error Handling Middleware**។

វាជា Middleware ពិសេសមួយដែលមាន Parameter ៤ គឺ `(err, req, res, next)` ដែលត្រូវដាក់នៅខាងក្រោមគេបំផុតនៃ Route ទាំងអស់។

```javascript
const express = require('express');
const app = express();

// Route ធម្មតា
app.get('/api/users', (req, res, next) => {
    try {
        // ឧបមាថាយើងព្យាយាមហៅមុខងារដែលមិនមាន
        undefinedFunction(); 
    } catch (error) {
        // បញ្ជូនកំហុសទៅកាន់ Error Handler
        next(error);
    }
});

// ត្រូវដាក់នៅខាងក្រោមគេបំផុត
// Global Error Handler
app.use((err, req, res, next) => {
    console.error(err.stack); // បង្ហាញកំហុសក្នុង Terminal សម្រាប់ Developer មើល
    
    // ឆ្លើយតបទៅកាន់ User វិញជាមួយនឹង Status 500
    res.status(500).json({
        success: false,
        message: "មានបញ្ហាបច្ចេកទេសនៅលើ Server (Internal Server Error)"
    });
});
```

## 🛡️ ការត្រួតពិនិត្យទិន្នន័យ (Data Validation)

កុំជឿជាក់លើទិន្នន័យដែលអ្នកប្រើប្រាស់បញ្ជូនមក! មុននឹងយកទិន្នន័យពី `req.body` ទៅរក្សាទុកក្នុង Database អ្នកត្រូវតែពិនិត្យមើលសិនថា តើវាមានទម្រង់ត្រឹមត្រូវដែរឬទេ? (ឧ. តើ Email មានសញ្ញា @ ទេ? តើលេខសម្ងាត់មានប្រវែងលើសពី ៦ តួអក្សរទេ?)

បើយើងប្រើ `if/else` ធម្មតាដើម្បីពិនិត្យ វានឹងស៊ីពេលនិងកូដវែងឆ្ងាយ។ ដូច្នេះហើយ ទើបគេនិយមប្រើ **Zod** ឬ **Joi**។

### ការប្រើប្រាស់ Zod

**Zod** គឺជាបណ្ណាល័យ (Library) ដ៏ពេញនិយមបំផុតបច្ចុប្បន្នសម្រាប់ធ្វើ Validation ជាពិសេសដោយសារតែវាស៊ីគ្នាជាមួយ TypeScript ខ្លាំង។

**១. ដំឡើង Zod:**
```bash
npm install zod
```

**២. បង្កើត Schema និង Validate:**
```javascript
const { z } = require('zod');

// ១. កំណត់លក្ខខណ្ឌ (Schema)
const userSchema = z.object({
  username: z.string().min(3, "ឈ្មោះត្រូវមានយ៉ាងហោចណាស់ ៣តួអក្សរ"),
  email: z.string().email("ទម្រង់អុីមែលមិនត្រឹមត្រូវ"),
  age: z.number().min(18, "ត្រូវមានអាយុចាប់ពី ១៨ឆ្នាំឡើងទៅ").optional() // .optional() មានន័យថាមិនចាំបាច់មានក៏បាន
});

app.post('/api/register', (req, res) => {
    try {
        // ២. យកទិន្នន័យមកផ្ទៀងផ្ទាត់
        const validData = userSchema.parse(req.body);
        
        // បើត្រឹមត្រូវ វានឹងបន្តមកដល់ទីនេះ
        res.status(201).json({ message: "ចុះឈ្មោះជោគជ័យ", data: validData });
        
    } catch (error) {
        // បើខុស វាធ្លាក់មកចូល catch 
        // យើងចាប់យកកំហុសពី Zod មកបង្ហាញជា JSON
        if (error instanceof z.ZodError) {
            return res.status(400).json({ 
                success: false, 
                errors: error.errors.map(e => e.message) 
            });
        }
        
        res.status(500).json({ message: "Internal Server Error" });
    }
});
```

ឥឡូវនេះ API របស់អ្នកមានសុវត្ថិភាពពីការបញ្ចូលទិន្នន័យមិនត្រឹមត្រូវ និងមិនងាយគាំង (Crash) ផ្តេសផ្តាសទៀតឡើយ!

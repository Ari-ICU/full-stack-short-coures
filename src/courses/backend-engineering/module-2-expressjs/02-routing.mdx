## 🚦 ការប្រើប្រាស់ Routing ក្នុង Express

Routing គឺជាការកំណត់ថា តើកម្មវិធីគួរតែឆ្លើយតបបែបណា នៅពេលដែលអតិថិជន (Client) ហៅទៅកាន់ Path ឬ URL ណាមួយ។

### ការចាប់យកទិន្នន័យពី Client

ពេលខ្លះ Client ត្រូវការបញ្ជូនទិន្នន័យមកកាន់ Server។ យើងអាចចាប់យកវាបានតាម ៣ វិធី៖
1. **Route Parameters (`req.params`):** ប្រើសម្រាប់បញ្ជូនលេខសម្គាល់ (ID) នៅក្នុង URL។
2. **Query Parameters (`req.query`):** ប្រើសម្រាប់ការស្វែងរក (Search) ឬ Filter។
3. **Request Body (`req.body`):** ប្រើសម្រាប់បញ្ជូនទិន្នន័យធំៗ (ដូចជាពេលបង្កើត User ថ្មី)។

```tabs
javascript
// routes.js
import express from 'express';
  const app = express();

  // ចាំបាច់ត្រូវមាន ដើម្បីអោយ Express អាចអានទិន្នន័យប្រភេទ JSON ពី req.body បាន
  app.use(express.json()); 

  // 1. Route Parameters (req.params)
  // ការហៅ: GET /users/123
  app.get('/users/:id', (req, res) => {
      const userId = req.params.id;
      res.json({ message: `កំពុងស្វែងរកទិន្នន័យរបស់អ្នកប្រើប្រាស់លេខ ${userId}` });
  });

  // 2. Query Parameters (req.query)
  // ការហៅ: GET /products?category=shoes&sort=price
  app.get('/products', (req, res) => {
      const { category, sort } = req.query;
      res.json({ category, sort, items: [] });
  });

  // 3. Request Body (req.body)
  // ការហៅ: POST /users (ផ្ញើ JSON Body មកជាមួយ)
  app.post('/users', (req, res) => {
      const { name, email } = req.body;
      res.status(201).json({ 
          message: "បង្កើត User បានជោគជ័យ!", 
          data: { name, email } 
      });
  });

  app.listen(3000, () => console.log('Server started on port 3000'));
```

### ការបំបែក Routes ជាឯកសារផ្សេងៗ (Express Router)

នៅពេលកម្មវិធីធំទៅៗ យើងមិនអាចសរសេរ Routes ទាំងអស់នៅក្នុង `app.js` ទេ។ យើងត្រូវប្រើ `express.Router()`។

```tabs
javascript
// app.js
import express from 'express';
  import userRoutes from './routes/userRoutes.js';

  const app = express();
  app.use(express.json());

  // ភ្ជាប់ Router
  app.use('/api/users', userRoutes);

  app.listen(3000);
---
javascript
// routes/userRoutes.js
import express from 'express';
  const router = express.Router();

  // Route នេះមាន Path ពេញគឺ /api/users
  router.get('/', (req, res) => {
      res.json([{ id: 1, name: "Sok" }]);
  });

  // Route នេះមាន Path ពេញគឺ /api/users/:id
  router.get('/:id', (req, res) => {
      res.json({ id: req.params.id, name: "Sok" });
  });

  export default router;
```

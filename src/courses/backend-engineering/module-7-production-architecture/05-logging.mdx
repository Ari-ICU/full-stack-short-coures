## 📝 ហេតុអ្វីយើងមិនគួរប្រើត្រឹម `console.log`?

ពេលអ្នក Deploy កម្មវិធីទៅលើ Server ការប្រើប្រាស់ `console.log()` ធម្មតាគឺមិនគ្រប់គ្រាន់ទេ ព្រោះ៖
1. វាពិបាកក្នុងការរុករកកំហុស (គ្មានម៉ោងពិតប្រាកដ)។
2. វាមិនអាចរក្សាទុក Logs ទៅក្នុង File អោយយើងបានទេ។
3. យើងមិនអាចបែងចែកកម្រិតនៃកំហុស (info, warn, error) បានច្បាស់លាស់។

យើងតែងតែប្រើប្រាស់ Library ឈ្មោះ **Winston** ឫ **Pino**។

### ការប្រើប្រាស់ Winston

```bash
npm install winston
```

```tabs
javascript
// utils/logger.js
import winston from 'winston';

export const logger = winston.createLogger({
    level: process.env.NODE_ENV === 'production' ? 'info' : 'debug',
    format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.json() // បម្លែងជា JSON ដើម្បីអោយប្រព័ន្ធស្រូប Logs អានបានងាយ
    ),
    transports: [
        // បង្ហាញលើ Terminal ពេល Development
        new winston.transports.Console({
            format: winston.format.simple()
        }),
        // រក្សាទុកកំហុសទៅក្នុងឯកសារ (សម្រាប់ Production)
        new winston.transports.File({ filename: 'logs/error.log', level: 'error' }),
        new winston.transports.File({ filename: 'logs/combined.log' })
    ]
});
---
javascript
// app.js (ឧទាហរណ៍នៃការយកទៅប្រើ)
import { logger } from './utils/logger.js';

app.use((err, req, res, next) => {
    // កត់ត្រាកំហុសចូលទៅក្នុងឯកសារ error.log
    logger.error(`មានកំហុស: ${err.message}`, { url: req.url, method: req.method });
    res.status(500).json({ error: "Server Error" });
});
```

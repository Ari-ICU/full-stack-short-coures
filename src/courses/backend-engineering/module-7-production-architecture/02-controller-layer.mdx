## 🕹️ តួនាទីរបស់ Controller

**Controller** មានតួនាទីប្រៀបដូចជាអ្នករត់តុ (Waiter) នៅក្នុងភោជនីយដ្ឋាន។ 
វាទទួលការបញ្ជាទិញ (Request) ពីអតិថិជន រួចយកទៅអោយចុងភៅ (Service Layer) ធ្វើ។ ពេលចុងភៅធ្វើរួច វាជាអ្នកលើកម្ហូប (Response) ទៅអោយអតិថិជនវិញ។ 

**ច្បាប់មាស:** Controller **មិនត្រូវ** មានកូដសរសេរទាក់ទងនឹង Database ឬ ការគិតលេខស្មុគស្មាញឡើយ!

```tabs
javascript
// controllers/productController.js
import * as productService from '../services/productService.js';

export const createProduct = async (req, res, next) => {
    try {
        // 1. ទទួលទិន្នន័យពី Request
        const { name, price } = req.body;
        
        // 2. បញ្ជូនទិន្នន័យទៅ Service អោយធ្វើការ (គ្មានការសរសេរកូដ TypeORM ទីនេះទេ)
        const product = await productService.createNewProduct(name, price);
        
        // 3. បញ្ជូនលទ្ធផលទៅ Client វិញ
        res.status(201).json({ success: true, data: product });
    } catch (error) {
        // បញ្ជូន Error ទៅអោយ Global Error Handler
        next(error);
    }
};
```

## 🗄️ តួនាទីរបស់ Repository

**Repository Layer** មានតួនាទីតែមួយគត់គឺ ទាក់ទងជាមួយ Database។ 
រាល់កូដ TypeORM (ឬ Raw SQL) ទាំងអស់ត្រូវសរសេរនៅទីនេះ។ Service Layer គ្រាន់តែហៅមុខងារពី Repository ប៉ុណ្ណោះ វាមិនខ្វល់ថា Repository ប្រើ Postgres, MySQL, ឫ MongoDB នោះទេ។

ការបែងចែកនេះធ្វើអោយកម្មវិធីយើងងាយស្រួលក្នុងការធ្វើតេស្ត (Unit Testing) និងងាយស្រួលប្តូរ Database ទៅថ្ងៃក្រោយ។

```tabs
javascript
// repositories/productRepository.js
import { AppDataSource } from '../config/database.js';
import { Product } from '../entities/Product.js';

const repo = AppDataSource.getRepository(Product);

export const saveProduct = async (productData) => {
    const product = repo.create(productData);
    return await repo.save(product);
};

export const findProductById = async (id) => {
    return await repo.findOne({ where: { id } });
};

// Query ស្មុគស្មាញក៏ត្រូវសរសេរនៅទីនេះដែរ
export const findExpensiveProducts = async (minPrice) => {
    return await repo.createQueryBuilder("p")
        .where("p.price >= :price", { price: minPrice })
        .getMany();
};
```

## 🔐 ការរៀបចំប្រព័ន្ធសុវត្ថិភាព

នៅក្នុងការបង្កើតកម្មវិធីពិតប្រាកដ ការប្រើប្រាស់បណ្ណាល័យដូចជា **Auth.js** (ដែលពីមុនហៅថា NextAuth.js) គឺជាជម្រើសដ៏ពេញនិយម។ ប៉ុន្តែសូមចាំថា Auth.js មិនមែនជាវេទមន្តដែលការពារគ្រប់យ៉ាងដោយស្វ័យប្រវត្តិនោះទេ អ្នកនៅតែត្រូវយល់ពីរបៀបដែលសុវត្ថិភាពដំណើរការ។

## 1. Authentication (ការបញ្ជាក់អត្តសញ្ញាណ)

នេះគឺជាដំណើរការដែលបញ្ជាក់ថា **"អ្នកជានរណា"** តាមរយៈការវាយអុីមែល និងលេខសម្ងាត់ ឬការ Login តាមរយៈ Google/Facebook ជាដើម។ 

នៅពេលអ្នកប្រើប្រាស់ Login ជោគជ័យ Auth.js នឹងបង្កើត **Session** មួយ ហើយបញ្ជូន **Cookie** ទៅរក្សាទុកនៅក្នុង Browser របស់អ្នកប្រើប្រាស់។ Cookie នេះមានសុវត្ថិភាពខ្ពស់ (HTTP-only) ដែលការពារមិនអោយ Hacker លួចបានដោយងាយតាមរយៈ JavaScript នោះទេ។

```ts
// app/api/auth/[...nextauth]/route.ts
import NextAuth from "next-auth";
import GithubProvider from "next-auth/providers/github";

const handler = NextAuth({
  providers: [
    GithubProvider({
      clientId: process.env.GITHUB_ID as string,
      clientSecret: process.env.GITHUB_SECRET as string,
    }),
  ],
});

export { handler as GET, handler as POST };
```

## 2. ការទាញយក Session

បន្ទាប់ពី Login រួចរាល់ អ្នកអាចទាញយកព័ត៌មានអ្នកប្រើប្រាស់ (Session) នៅគ្រប់កន្លែងក្នុង Next.js។

```tsx
import { getServerSession } from "next-auth";

export default async function Dashboard() {
  const session = await getServerSession();

  if (!session) {
    return <h1>សូមចូលគណនីជាមុនសិន!</h1>;
  }

  return <h1>សូមស្វាគមន៍, {session.user?.name}!</h1>;
}
```

## 3. Authorization (ការផ្តល់សិទ្ធិ)

ការគ្រាន់តែ Login ចូលបាន មិនមែនមានន័យថាអ្នកប្រើប្រាស់អាចធ្វើអ្វីក៏បាននោះទេ។ នេះហៅថា **Authorization**។
ឧទាហរណ៍ អ្នកប្រហែលជាចង់ការពារកុំអោយ User ធម្មតាអាចចូលទំព័រ `/admin` បាន។ យើងធ្វើបែបនេះដោយពិនិត្យមើល **Roles** ឬ **Permissions**។

```tsx
export default async function AdminPage() {
  const session = await getServerSession();

  // 1. ពិនិត្យមើលថាមាន Login ឫនៅ (Authentication)
  if (!session) {
    return <h1>តម្រូវអោយ Login!</h1>;
  }

  // 2. ពិនិត្យមើលថាតើគាត់មានសិទ្ធិជា Admin ដែរឬទេ (Authorization)
  // (ឧបមាថាយើងបានកំណត់ role ទៅក្នុង session រួចរាល់)
  if (session.user?.role !== "admin") {
    return <h1>អ្នកគ្មានសិទ្ធិចូលទំព័រនេះទេ!</h1>;
  }

  return <h1>ទំព័រគ្រប់គ្រងសម្រាប់ Admin តែប៉ុណ្ណោះ 🔒</h1>;
}
```

ការប្រើប្រាស់ Database ដើម្បីចងចាំអ្នកប្រើប្រាស់ និងកំណត់សិទ្ធិអោយពួកគេ គឺជារឿងសំខាន់សម្រាប់សុវត្ថិភាពប្រព័ន្ធរបស់អ្នក!

## 🛡️ អ្វីទៅជា Middleware?

ស្រមៃថាមានអគារមួយ (គេហទំព័ររបស់អ្នក) ដែលមានបន្ទប់ជាច្រើន (ទំព័រផ្សេងៗ)។ 
នៅក្នុង React ធម្មតា ដើម្បីការពារបន្ទប់ អ្នកត្រូវយកសន្តិសុខទៅឈរយាមនៅមុខបន្ទប់នីមួយៗ (Protected Routes)។
ប៉ុន្តែ **Middleware** ប្រៀបដូចជាការដាក់សន្តិសុខនៅក្លោងទ្វារធំនៃអគារតែម្តង! មុននឹងអ្នកប្រើប្រាស់អាចបើកទំព័រណាមួយ ឬហៅ API ណាមួយបាន ពួកគេត្រូវឆ្លងកាត់ការត្រួតពិនិត្យរបស់ Middleware ជាមុនសិន។

នេះគឺជាបច្ចេកទេសដ៏អស្ចារ្យសម្រាប់ការធ្វើ Authentication, ការប្តូរភាសា (i18n), និងការកែប្រែ Headers!

## 1. ការបង្កើត Middleware

នៅក្នុងគម្រោងដែលប្រើប្រាស់ថត `src` អ្នកត្រូវបង្កើតឯកសារឈ្មោះ `middleware.ts` នៅខាងក្នុង `src` (នៅស្មើនឹងថត `app` មិនមែននៅក្នុងថត `app` ទេ)។

**ឯកសារ `src/middleware.ts`:**
```ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

// មុខងារនេះនឹងរត់រាល់ពេលដែលមានគេចូលមើលគេហទំព័រអ្នក
export function middleware(request: NextRequest) {
  console.log("មានគេកំពុងសុំចូលទៅកាន់:", request.nextUrl.pathname);
  
  // បើកទ្វារអោយចូលដោយសេរី!
  return NextResponse.next();
}
```

## 2. ការការពារទំព័រដោយប្រើ Middleware (Authentication)

ឧទាហរណ៍ថាយើងចង់ការពាររាល់ទំព័រទាំងអស់ដែលផ្តើមដោយ `/dashboard`។ បើគេគ្មាន Token ទេ យើងនឹងដេញគេទៅទំព័រ `/login` វិញ។

```ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  // ១. ចាប់យកទីតាំងដែលគេកំពុងចង់ទៅ
  const path = request.nextUrl.pathname;

  // ២. ឆែកមើលតើវាជាទំព័រឯកជនដែរឬទេ? (ឧ. /dashboard ឬ /dashboard/settings)
  const isProtectedPath = path.startsWith('/dashboard');

  // ៣. ទាញយក Token (ឧបមាថាយើងទុកវាក្នុង Cookies)
  const token = request.cookies.get('myToken')?.value;

  // ៤. សម្រេចចិត្ត
  if (isProtectedPath && !token) {
    // គ្មានសិទ្ធិទេ! ដេញទៅទំព័រ /login វិញចុះ
    return NextResponse.redirect(new URL('/login', request.url));
  }

  if (path === '/login' && token) {
    // បើមាន Token ហើយ តែចង់ចូលទំព័រ /login ទៀត? ដេញទៅ /dashboard វិញចុះ!
    return NextResponse.redirect(new URL('/dashboard', request.url));
  }

  // អោយចូលតាមសម្រួល
  return NextResponse.next();
}
```

## 3. ការកំណត់ដែនយាមកាម (Matcher)

ជាធម្មតា Middleware ខាងលើនឹងរត់គ្រប់ពេល! ទោះបីជាពេលទាញយករូបភាព (`.png`) ឬឯកសារ CSS ក៏វារត់ដែរ ដែលធ្វើអោយខាតពេល។ ដើម្បីប្រាប់ Middleware ថាកុំអោយរត់នៅទំព័រខ្លះ យើងត្រូវសរសេរកូដ `config`។

បញ្ចូលកូដនេះទៅខាងក្រោមចុងបញ្ចាប់នៃឯកសារ `middleware.ts`៖

```ts
// កំណត់ថាអោយ Middleware រត់តែពេលចូលទំព័រណាខ្លះ
export const config = {
  matcher: [
    /*
     * អោយវារត់គ្រប់ទំព័រទាំងអស់ លើកលែងតែ៖
     * - api (API routes)
     * - _next/static (ឯកសារកូដដែលបាន compile រួច)
     * - _next/image (ឯកសាររូបភាព)
     * - favicon.ico (រូប Logo តូចនៅខាងលើ Browser)
     */
    '/((?!api|_next/static|_next/image|favicon.ico).*)',
  ],
};
```

ការប្រើប្រាស់ Middleware ធ្វើអោយគេហទំព័ររបស់អ្នកមានសុវត្ថិភាពខ្ពស់ និងល្បឿនលឿន ព្រោះការរារាំងអ្នកដែលគ្មានសិទ្ធិត្រូវបានធ្វើឡើងនៅកម្រិត Server ដោយមិនចាំបាច់រង់ចាំអោយ Browser ទាញយកកូដ Component នោះទេ។

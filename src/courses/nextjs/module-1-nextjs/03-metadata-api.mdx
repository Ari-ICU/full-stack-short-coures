## 🔍 សារៈសំខាន់នៃ SEO

**SEO (Search Engine Optimization)** គឺជាការរៀបចំគេហទំព័ររបស់អ្នកអោយស្របតាមស្តង់ដាររបស់ Google ដើម្បីអោយគេហទំព័រអ្នកអាចលោតនៅលេខរៀងលើគេពេលមានគេស្វែងរក។ 

ចំណុចដ៏សំខាន់បំផុតមួយនៃ SEO គឺការប្រាប់ទៅកាន់ Google ថា "តើទំព័រនេះនិយាយពីអ្វី?" តាមរយៈ **Metadata** ដូចជា ចំណងជើង (Title) និងការពណ៌នា (Description) ដែលលាក់ទុកនៅក្នុងកូដ `<head>`។ 

នៅក្នុង React ធម្មតា យើងត្រូវដំឡើង Library ដូចជា `react-helmet` យ៉ាងស្មុគស្មាញ។ ប៉ុន្តែនៅក្នុង Next.js ការងារនេះគឺងាយស្រួលបំផុត!

## 1. ការកំណត់ Metadata ថេរ (Static Metadata)

អ្នកគ្រាន់តែ Export អថេរមួយឈ្មោះថា `metadata` ចេញពីឯកសារ `layout.tsx` ឬ `page.tsx` ណាមួយ នោះ Next.js នឹងរៀបចំវាទៅជា HTML យ៉ាងត្រឹមត្រូវដោយស្វ័យប្រវត្តិ។

**ឯកសារ `app/about/page.tsx`:**
```tsx
import type { Metadata } from 'next';

// វានឹងយកចំណងជើងនេះទៅដាក់ក្នុងផ្ទាំងខាងលើរបស់ Browser និងក្នុង Google
export const metadata: Metadata = {
  title: 'អំពីក្រុមហ៊ុនយើង - My Store',
  description: 'ស្វែងយល់បន្ថែមពីប្រវត្តិ និងគោលដៅរបស់ក្រុមហ៊ុន My Store ក្នុងការផ្តល់សេវាកម្មដ៏ល្អបំផុត។',
};

export default function AboutPage() {
  return (
    <div>
      <h1>អំពីយើង</h1>
      <p>សូមស្វាគមន៍មកកាន់ក្រុមហ៊ុនរបស់យើង...</p>
    </div>
  );
}
```

## 2. ការកំណត់ Metadata ឆ្លាតវៃ (Dynamic Metadata)

ឧបមាថាអ្នកមានទំព័រលក់ទំនិញ (Product Page) ដែលមានទំនិញរាប់ពាន់មុខ។ អ្នកមិនអាចសរសេរ Metadata ថេរសម្រាប់ទំនិញទាំងពាន់នោះបានទេ។ អ្នកចង់អោយចំណងជើងទំព័រប្តូរទៅតាមឈ្មោះទំនិញដែលគេកំពុងមើល!

ដើម្បីធ្វើបែបនេះបាន អ្នកត្រូវប្រើប្រាស់អនុគមន៍ **`generateMetadata`**។

**ឯកសារ `app/products/[id]/page.tsx`:**
```tsx
import type { Metadata } from 'next';

type Props = {
  params: { id: string }
};

// 1. បង្កើតអនុគមន៍នេះដើម្បីទាញយកទិន្នន័យមកធ្វើជា Metadata
export async function generateMetadata({ params }: Props): Promise<Metadata> {
  // ឧបមាថាយើងហៅ API ទៅ Server ដើម្បីយកព័ត៌មានទំនិញដោយផ្អែកលើ ID
  const product = await fetch(`https://api.example.com/products/${params.id}`).then((res) => res.json());

  return {
    title: `${product.name} | My Store`, // ឧទាហរណ៍ "កាបូបស្បែក | My Store"
    description: product.description,
    // អ្នកក៏អាចកំណត់រូបភាពសម្រាប់ពេលគេចែករំលែកចូល Facebook ផងដែរ (Open Graph)
    openGraph: {
      images: [product.imageUrl],
    },
  };
}

// 2. នេះគឺជា Component សម្រាប់បង្ហាញលើអេក្រង់ធម្មតា
export default function ProductPage({ params }: Props) {
  return (
    <div>
      <h1>ទំនិញលេខ: {params.id}</h1>
      {/* ទីនេះជាកន្លែងបង្ហាញទិន្នន័យទំនិញ... */}
    </div>
  );
}
```

> 💡 **ចំណាំ:** Next.js មានភាពឆ្លាតវៃណាស់! ទោះបីជាអ្នកហៅ API `fetch(...)` ដើម្បីយកទិន្នន័យដូចគ្នានៅក្នុង `generateMetadata` ម្តង និងនៅក្នុង `ProductPage` ម្តងទៀតក៏ដោយ ក៏វានឹងមិនធ្វើការហៅ API នោះពីរដងដែលធ្វើអោយយឺតនោះទេ។ វាហៅតែម្តងហើយចងចាំទុក (Deduplication)!

# ⚠️ Error State & Retry Mechanics

កំហុសបណ្តាញ (Network Glitches, Server Down, Validation Failures) គឺជារឿងចៀសមិនផុត។ React App ដែលរឹងមាំត្រូវតែមាន Error UI ច្បាស់លាស់។

---

## ឧទាហរណ៍៖ Error Card ជាមួយ Retry Button

```tsx
interface ErrorCardProps {
  message?: string;
  onRetry: () => void;
}

export function ErrorCard({ message = "មានបញ្ហាបច្ចេកទេសក្នុងការទាញយកទិន្នន័យ", onRetry }: ErrorCardProps) {
  return (
    <div className="p-6 bg-rose-500/10 border border-rose-500/20 rounded-2xl text-center space-y-4 max-w-md mx-auto">
      <div className="w-12 h-12 bg-rose-500/20 text-rose-400 rounded-full flex items-center justify-center mx-auto text-xl font-bold">
        !
      </div>
      <div>
        <h3 className="text-base font-semibold text-white">បរាជ័យក្នុងការតភ្ជាប់</h3>
        <p className="text-sm text-rose-300 mt-1">{message}</p>
      </div>
      <button
        onClick={onRetry}
        className="px-4 py-2 bg-rose-600 hover:bg-rose-500 text-white text-sm font-medium rounded-lg transition"
      >
        ព្យាយាមម្តងទៀត (Retry)
      </button>
    </div>
  );
}
```

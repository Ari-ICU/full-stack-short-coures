## 🔐 លំហូរការងារនៃការ Login (Authentication Flow)

ការធ្វើ Authentication មិនមែនជាការងាររបស់ Frontend សុទ្ធសាធនោះទេ។ វាគឺជាការសហការគ្នាយ៉ាងជិតស្និទ្ធរវាង React និង Backend (ឧ. Node.js, Laravel)។

```mermaid
flowchart TD
    A["អ្នកប្រើប្រាស់បញ្ចុល Email/Password"] -->|fetch(POST)| B("Backend ពិនិត្យភាពត្រឹមត្រូវ")
    B -- បរាជ័យ --> C["React បង្ហាញពាក្យ 'លេខសម្ងាត់ខុស'"]
    B -- ជោគជ័យ --> D["Backend ផ្តល់អោយនូវ Token (កូនសោរ)"]
    D --> E["React រក្សាទុក Token ក្នុង localStorage"]
    E --> F["React ប្តូរ State ទៅជា 'បាន Login'"]
    F --> G["ប្តូរទំព័រទៅកាន់ Dashboard"]
```

## 💾 ការរក្សាទុក Token និង Auth State

ដើម្បីដឹងថាអ្នកប្រើប្រាស់បាន Login រួចឬនៅ យើងត្រូវបង្កើត State មួយ (ជាទូទៅគេដាក់វានៅកម្រិតខ្ពស់ដូចជា Context ដើម្បីចែករំលែកគ្រប់ទំព័រ)។

```jsx
import { useState, useEffect } from 'react';

function AuthApp() {
  const [user, setUser] = useState(null);

  // 1. ពេលទំព័រ Load ដំបូង ឆែកមើលថាតើមាន Token ពីមុនដែរឬទេ?
  useEffect(() => {
    const token = localStorage.getItem('token');
    if (token) {
      // (បើជាការងារពិត អ្នកត្រូវហៅ API ទៅឆែកថាតើ Token នេះហួសកំណត់ឫនៅសិន)
      setUser({ name: "រដ្ឋា", role: "admin" }); 
    }
  }, []);

  const handleLogin = async (e) => {
    e.preventDefault();
    try {
      // ឧបមាថាការហៅ API ជោគជ័យ...
      const mockToken = "abcdef123456";
      
      // 2. រក្សាទុកកូនសោរចូលក្នុងឃ្លាំង Browser
      localStorage.setItem('token', mockToken);
      
      // 3. ផ្លាស់ប្តូរ State
      setUser({ name: "រដ្ឋា", role: "admin" });
      
      alert("ចូលគណនីបានជោគជ័យ!");
    } catch (err) {
      console.error(err);
    }
  };

  const handleLogout = () => {
    // 4. ការ Logout គឺគ្រាន់តែលុបកូនសោរ និង State ចោល!
    localStorage.removeItem('token');
    setUser(null);
  };

  return (
    <div>
      {user ? (
        <>
          <h1>សួស្តី {user.name}!</h1>
          <button onClick={handleLogout}>ចេញពីគណនី</button>
        </>
      ) : (
        <form onSubmit={handleLogin}>
          {/* ប្រអប់វាយ Email & Password */}
          <button type="submit">ចូលគណនី</button>
        </form>
      )}
    </div>
  );
}
```

> **⚡ សំខាន់:** រាល់ពេលដែលអ្នកហៅ API ទៅយកទិន្នន័យឯកជន (ឧ. ពត៌មានគណនីធនាគារ) ក្រោយពេល Login រួច អ្នកត្រូវតែបញ្ជូន Token នោះទៅវិញតាមរយៈ Header `Authorization: Bearer <token>` ដើម្បីប្រាប់ Backend ថាអ្នកជានរណា!

# 🔀 Conditional Rendering Patterns

នៅក្នុង React យើងអាចប្រើ JavaScript Logic ធម្មតា (ដូចជា `if/else`, Ternary, `&&`) ដើម្បីសម្រេចថាតើត្រូវ Render Component ណាលើ Screen។

---

## ១. Early Return Pattern (ល្អបំផុតសម្រាប់ Loading & Errors)

```tsx
interface UserDashboardProps {
  isLoading: boolean;
  error: string | null;
  user: { name: string; email: string } | null;
}

export function UserDashboard({ isLoading, error, user }: UserDashboardProps) {
  if (isLoading) {
    return <div className="p-8 text-center text-slate-400 animate-pulse">កំពុងទាញយកទិន្នន័យ...</div>;
  }

  if (error) {
    return (
      <div className="p-6 bg-rose-500/10 border border-rose-500/20 rounded-xl text-rose-400">
        <p className="font-semibold">បរាជ័យក្នុងការទាញយក</p>
        <p className="text-sm">{error}</p>
      </div>
    );
  }

  if (!user) {
    return <div className="p-8 text-slate-400">រកមិនឃើញគណនីអ្នកប្រើប្រាស់ឡើយ។</div>;
  }

  return (
    <div className="p-6 bg-slate-900 rounded-xl text-white">
      <h2 className="text-xl font-bold">{user.name}</h2>
      <p className="text-slate-400">{user.email}</p>
    </div>
  );
}
```

## ២. Ternary Operator (`? :`) & Short-Circuit (`&&`)

```tsx
export function StatusBanner({ isOnline, unreadCount }: { isOnline: boolean; unreadCount: number }) {
  return (
    <div className="p-4 bg-slate-900 rounded-xl flex items-center justify-between">
      <span className={isOnline ? "text-emerald-400" : "text-slate-500"}>
        {isOnline ? "● កំពុង Online" : "○ កំពុង Offline"}
      </span>

      {/* ⚠️ ប្រយ័ត្ន៖ ប្រើ unreadCount > 0 ជៀសវាងលេខ 0 លេចលើ Screen */}
      {unreadCount > 0 && (
        <span className="px-2 py-0.5 bg-rose-600 text-white text-xs rounded-full">
          {unreadCount} សារថ្មី
        </span>
      )}
    </div>
  );
}
```

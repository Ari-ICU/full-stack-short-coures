## 🎯 គោលដៅ Mini Project

យើងនឹងបង្កើត **Dynamic Product Catalog** ដែលអនុញ្ញាតឱ្យបង្ហាញផលិតផលជាច្រើន អមដោយស្លាកសញ្ញា (Badges) បង្ហាញពីស្ថានភាពស្តុក (In Stock, Low Stock, Out of Stock) ដោយស្វ័យប្រវត្តិតាមរយៈ Conditional Rendering។

## 🛠️ ការសរសេរកូដអនុវត្តជាក់ស្តែង

### 1. បង្កើត StockBadge Component (`src/components/StockBadge.jsx`)
```jsx
export function StockBadge({ quantity }) {
  if (quantity === 0) {
    return (
      <span className="bg-red-100 text-red-700 text-xs font-bold px-2.5 py-1 rounded-full">
        អស់ពីស្តុក (Out of Stock)
      </span>
    );
  }

  if (quantity < 5) {
    return (
      <span className="bg-amber-100 text-amber-800 text-xs font-bold px-2.5 py-1 rounded-full">
        នៅសល់តែ {quantity} គ្រឿងទៀតប៉ុណ្ណោះ!
      </span>
    );
  }

  return (
    <span className="bg-emerald-100 text-emerald-800 text-xs font-bold px-2.5 py-1 rounded-full">
      មានក្នុងស្តុក ({quantity})
    </span>
  );
}
```

### 2. បង្កើត ProductCard Component (`src/components/ProductCard.jsx`)
```jsx
import { StockBadge } from './StockBadge';

export default function ProductCard({ product }) {
  const { title, price, category, quantity, imageUrl, rating } = product;

  return (
    <div className="bg-white rounded-2xl shadow-sm hover:shadow-md transition border border-gray-100 p-5 flex flex-col justify-between">
      <div>
        <img src={imageUrl} alt={title} className="w-full h-44 object-cover rounded-xl mb-4" />
        <div className="flex justify-between items-center mb-2">
          <span className="text-xs font-semibold text-gray-500 uppercase">{category}</span>
          <StockBadge quantity={quantity} />
        </div>
        <h3 className="font-bold text-gray-900 text-lg line-clamp-1">{title}</h3>
        {rating && (
          <p className="text-sm text-amber-500 mt-1">★ {rating} / 5.0</p>
        )}
      </div>

      <div className="flex items-center justify-between mt-6 pt-4 border-t border-gray-50">
        <span className="text-2xl font-black text-gray-900">${price.toFixed(2)}</span>
        <button 
          disabled={quantity === 0}
          className={`px-4 py-2 rounded-xl font-bold transition text-sm ${
            quantity > 0 
              ? 'bg-blue-600 text-white hover:bg-blue-700' 
              : 'bg-gray-200 text-gray-400 cursor-not-allowed'
          }`}
        >
          {quantity > 0 ? 'ទិញឥឡូវនេះ' : 'មិនអាចទិញបាន'}
        </button>
      </div>
    </div>
  );
}
```

## 🪝 2. Composing Custom Hooks in React

### 💻 Code សម្រាប់តែ Custom Hooks Integration:

```jsx
// 1. Debounce user search input
const [query, setQuery] = useState('');
const debouncedQuery = useDebounce(query, 400);

// 2. Fetch API products
const { data: products, isLoading, error } = useFetch('https://fakestoreapi.com/products');

// 3. Persistent Wishlist ក្នុង LocalStorage
const [wishlist, setWishlist] = useLocalStorage('user-wishlist', []);

// Toggle Wishlist Item
const toggleWishlist = (id) => {
  setWishlist((prev) =>
    prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id]
  );
};
```

### 🔍 ការពន្យល់មួយជួរៗ (Line-by-Line Explanation):

1. `useDebounce(query, 400)`
   - ពន្យារការ Filter និង Fetch រហូតទាល់តែ User ឈប់វាយ។
2. `useFetch('https://fakestoreapi.com/products')`
   - គ្រប់គ្រង State `products`, `isLoading`, និង `error` ដោយស្វ័យប្រវត្តិ។
3. `useLocalStorage('user-wishlist', [])`
   - រក្សាទុក IDs នៃផលិតផលដែល User ចុច ❤️ ក្នុង Storage មិនបាត់បង់ពេល Refresh។

## 🚀 3. Complete Integrated Product Search Project

```jsx
import { useState } from 'react';
import { useFetch } from '../hooks/useFetch';
import { useDebounce } from '../hooks/useDebounce';
import { useLocalStorage } from '../hooks/useLocalStorage';

export default function ProductSearchApp() {
  const [query, setQuery] = useState('');
  const debouncedQuery = useDebounce(query, 400);

  const { data: products, isLoading, error } = useFetch('https://fakestoreapi.com/products');
  const [wishlist, setWishlist] = useLocalStorage('saved-wishlist', []);

  const toggleWishlist = (id) => {
    setWishlist((prev) =>
      prev.includes(id) ? prev.filter((item) => item !== id) : [...prev, id]
    );
  };

  const filtered = (products || []).filter((p) =>
    p.title.toLowerCase().includes(debouncedQuery.toLowerCase())
  );

  return (
    <div className="max-w-4xl mx-auto p-6">
      <div className="flex justify-between items-center mb-6">
        <h2 className="text-2xl font-bold text-slate-800 dark:text-white">🛍️ Product Catalog</h2>
        <span className="text-xs bg-red-100 text-red-600 px-3 py-1 rounded-full font-bold">
          ❤️ Wishlist ({wishlist.length})
        </span>
      </div>

      {/* SEARCH BAR */}
      <input
        type="text"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        placeholder="ស្វែងរកផលិតផល..."
        className="w-full px-4 py-2 border rounded-xl mb-6 dark:bg-slate-800 dark:text-white text-sm"
      />

      {/* STATES */}
      {isLoading && <p className="text-center py-8 text-slate-400">⏳ កំពុងទាញយកទិន្នន័យ...</p>}
      {error && <p className="text-center py-8 text-red-500">⚠️ {error}</p>}

      {/* GRID */}
      <div className="grid grid-cols-2 md:grid-cols-3 gap-4">
        {filtered.map((item) => {
          const isFavorited = wishlist.includes(item.id);
          return (
            <div key={item.id} className="p-4 bg-white dark:bg-slate-800 rounded-2xl shadow border flex flex-col justify-between">
              <div>
                <h4 className="font-bold text-sm text-slate-800 dark:text-white line-clamp-1">{item.title}</h4>
                <p className="text-blue-600 font-black mt-1">${item.price}</p>
              </div>
              <button
                onClick={() => toggleWishlist(item.id)}
                className={`mt-4 py-1.5 rounded-xl text-xs font-bold transition ${
                  isFavorited ? 'bg-red-500 text-white' : 'bg-slate-100 text-slate-700 hover:bg-slate-200'
                }`}
              >
                {isFavorited ? '❤️ Favorited' : '🤍 Add to Wishlist'}
              </button>
            </div>
          );
        })}
      </div>
    </div>
  );
}
```

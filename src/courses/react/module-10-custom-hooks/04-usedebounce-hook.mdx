## ⏱️ ១. Mental Model: Debounce vs Throttle

```
─── DEBOUNCE (រង់ចាំឈប់សកម្មភាពទើបដំណើរការ) ───
User types:  [R] -> [e] -> [a] -> [c] -> [t]
Timer:       (400ms restarted 5 times...) ───(User ឈប់វាយ 400ms)───> [🔥 EXECUTE FETCH 'React']

─── THROTTLE (ដំណើរការទៀងទាត់តាមចន្លោះពេលថេរ) ───
User scrolls: [~~~~~~~~~~ Scrolling Non-stop 2000ms ~~~~~~~~~~]
Execution:    [🔥 Execute at 0ms] ──(500ms)──> [🔥 Execute at 500ms] ──(500ms)──> [🔥 Execute at 1000ms]
```

## 💻 ២. កូដពេញលេញនៃ `src/hooks/useDebounce.js`

```javascript
// src/hooks/useDebounce.js
import { useState, useEffect } from 'react';

export function useDebounce(value, delay = 400) {
  // ១. State រក្សាទុកតម្លៃដែលបានពន្យារពេលរួច
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    // ២. បង្កើត Timer ដើម្បី Update តម្លៃក្រោយ delay ms
    const timer = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    // ៣. 🧹 CLEANUP: លុប Timer ចាស់ចោល ប្រសិនបើ value ផ្លាស់ប្តូរមុនពេល 400ms ចប់
    return () => {
      clearTimeout(timer);
    };
  }, [value, delay]);

  return debouncedValue;
}
```

## ⚡ ៣. បន្ថែម ៖ កូដនៃ `src/hooks/useThrottle.js` (សម្រាប់ Scroll/Resize)

```javascript
// src/hooks/useThrottle.js
import { useState, useEffect, useRef } from 'react';

export function useThrottle(value, interval = 500) {
  const [throttledValue, setThrottledValue] = useState(value);
  const lastExecuted = useRef(Date.now());

  useEffect(() => {
    if (Date.now() >= lastExecuted.current + interval) {
      lastExecuted.current = Date.now();
      setThrottledValue(value);
    } else {
      const timer = setTimeout(() => {
        lastExecuted.current = Date.now();
        setThrottledValue(value);
      }, interval);

      return () => clearTimeout(timer);
    }
  }, [value, interval]);

  return throttledValue;
}
```

```jsx
// src/components/LiveSearchBenchmark.jsx
import { useState, useEffect } from 'react';
import { useDebounce } from '../hooks/useDebounce';

export default function LiveSearchBenchmark() {
  const [keystrokes, setKeystrokes] = useState('');
  const debouncedSearch = useDebounce(keystrokes, 400);

  const [rawKeyCount, setRawKeyCount] = useState(0);
  const [networkCalls, setNetworkCalls] = useState(0);

  const handleTyping = (e) => {
    setKeystrokes(e.target.value);
    setRawKeyCount((c) => c + 1);
  };

  useEffect(() => {
    if (debouncedSearch.trim()) {
      // ក្លែងធ្វើការបាញ់ HTTP Request
      setNetworkCalls((c) => c + 1);
    }
  }, [debouncedSearch]);

  const savings =
    rawKeyCount > 0
      ? Math.round(((rawKeyCount - networkCalls) / rawKeyCount) * 100)
      : 0;

  return (
    <div className="max-w-md mx-auto p-6 bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-800 rounded-3xl shadow-xl space-y-5">
      <div>
        <h3 className="font-bold text-base text-slate-800 dark:text-white flex items-center gap-2">
          <span>⚡</span> Live Search Debounce Benchmark
        </h3>
        <p className="text-xs text-slate-500">សង្កេតមើលការកាត់បន្ថយ Network Requests ផ្ទាល់ភ្នែក</p>
      </div>

      <input
        type="text"
        value={keystrokes}
        onChange={handleTyping}
        placeholder="សាកល្បងវាយអក្សរឱ្យលឿននៅទីនេះ..."
        className="w-full px-4 py-2.5 border rounded-2xl text-xs dark:bg-slate-800 dark:text-white focus:outline-none focus:ring-2 focus:ring-indigo-500"
      />

      {/* BENCHMARK STATS */}
      <div className="grid grid-cols-3 gap-2 text-center text-xs">
        <div className="p-3 bg-slate-50 dark:bg-slate-800 rounded-2xl border border-slate-100 dark:border-slate-700">
          <span className="block text-[10px] text-slate-400 font-bold uppercase">ចុច Keyboard</span>
          <span className="font-black text-sm text-slate-800 dark:text-white">{rawKeyCount} ដង</span>
        </div>
        <div className="p-3 bg-slate-50 dark:bg-slate-800 rounded-2xl border border-slate-100 dark:border-slate-700">
          <span className="block text-[10px] text-slate-400 font-bold uppercase">បាញ់ API</span>
          <span className="font-black text-sm text-indigo-600 dark:text-indigo-400">{networkCalls} ដង</span>
        </div>
        <div className="p-3 bg-emerald-50 dark:bg-emerald-950/40 rounded-2xl border border-emerald-100 dark:border-emerald-900">
          <span className="block text-[10px] text-emerald-600 font-bold uppercase">សន្សំបាន</span>
          <span className="font-black text-sm text-emerald-600">{savings}%</span>
        </div>
      </div>
    </div>
  );
}
```

## 🎯 គោលដៅ Mini Project

យើងនឹងបង្កើត **High-Performance Data Grid** ដែលមានសមត្ថភាព៖
1. Filtering & Sorting លើ 1,000 ទិន្នន័យដោយគ្មាន Lag
2. Memoized Rows (`React.memo`)
3. Global Keyboard Shortcut (`Cmd+K` / `Ctrl+K`) ដើម្បី Auto-focus Search Box ដោយប្រើ `useRef`

---

## 🛠️ កូដពេញលេញនៃគម្រោង (`src/App.jsx`)

```jsx
import { useState, useMemo, useCallback, useRef, useEffect, memo } from 'react';

// Generate 1,000 Fake Data Records
const BIG_DATA = Array.from({ length: 1000 }, (_, i) => ({
  id: `USR-${i + 1}`,
  name: `User ${i + 1}`,
  email: `user${i + 1}@enterprise.com`,
  role: i % 3 === 0 ? 'Admin' : i % 2 === 0 ? 'Editor' : 'Viewer',
  score: Math.floor(Math.random() * 100),
}));

// 1. MEMOIZED ROW COMPONENT
const GridRow = memo(function GridRow({ item, isSelected, onSelect }) {
  return (
    <tr className={`border-b border-slate-100 hover:bg-blue-50/50 transition ${isSelected ? 'bg-blue-50' : ''}`}>
      <td className="p-3">
        <input
          type="checkbox"
          checked={isSelected}
          onChange={() => onSelect(item.id)}
          className="rounded text-blue-600 cursor-pointer"
        />
      </td>
      <td className="p-3 font-mono text-xs text-slate-500">{item.id}</td>
      <td className="p-3 font-bold text-slate-800 text-sm">{item.name}</td>
      <td className="p-3 text-slate-600 text-sm">{item.email}</td>
      <td className="p-3">
        <span className={`px-2 py-0.5 rounded-full text-xs font-bold ${
          item.role === 'Admin' ? 'bg-purple-100 text-purple-700' : 'bg-slate-100 text-slate-600'
        }`}>
          {item.role}
        </span>
      </td>
      <td className="p-3 font-bold text-slate-700">{item.score}</td>
    </tr>
  );
});

export default function OptimizedDataGrid() {
  const [query, setQuery] = useState('');
  const [selectedIds, setSelectedIds] = useState(new Set());
  const searchInputRef = useRef(null);

  // 2. KEYBOARD SHORTCUT (Cmd+K / Ctrl+K) FOCUS VIA useRef
  useEffect(() => {
    const handleKeyDown = (e) => {
      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
        e.preventDefault();
        searchInputRef.current?.focus();
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, []);

  // 3. useMemo FOR HEAVY FILTERING
  const filteredRows = useMemo(() => {
    if (!query) return BIG_DATA;
    return BIG_DATA.filter((row) =>
      row.name.toLowerCase().includes(query.toLowerCase()) ||
      row.email.toLowerCase().includes(query.toLowerCase()) ||
      row.role.toLowerCase().includes(query.toLowerCase())
    );
  }, [query]);

  // 4. useCallback FOR STABLE SELECTION HANDLER
  const handleToggleSelect = useCallback((id) => {
    setSelectedIds((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  }, []);

  return (
    <div className="min-h-screen bg-slate-50 p-8 max-w-5xl mx-auto">
      <div className="bg-white rounded-3xl p-8 shadow-sm border border-slate-100">
        
        {/* Header & Search Bar */}
        <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4 mb-6">
          <div>
            <h1 className="text-2xl font-black text-slate-800">⚡ High-Performance Data Grid</h1>
            <p className="text-slate-500 text-sm mt-1">បង្ហាញ 1,000 Rows ជាមួយ Memoization &amp; No Render Lag</p>
          </div>

          <div className="relative w-full sm:w-72">
            <input
              ref={searchInputRef}
              type="text"
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              placeholder="ស្វែងរក... (ចុច Ctrl+K)"
              className="w-full pl-4 pr-16 py-2.5 bg-slate-50 border border-slate-200 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
            />
            <kbd className="absolute right-3 top-2.5 bg-slate-200 text-slate-600 text-xs px-1.5 py-0.5 rounded font-mono">
              ⌘K
            </kbd>
          </div>
        </div>

        {/* Selection Stats */}
        <div className="flex justify-between items-center bg-slate-50 px-4 py-2 rounded-xl mb-4 text-xs font-bold text-slate-600">
          <span>បង្ហាញ៖ {filteredRows.length} ជួរ</span>
          <span>បានជ្រើសរើស៖ {selectedIds.size} នាក់</span>
        </div>

        {/* Data Table */}
        <div className="overflow-x-auto max-h-96 border border-slate-100 rounded-2xl">
          <table className="w-full text-left">
            <thead className="bg-slate-100/80 sticky top-0 backdrop-blur text-xs uppercase text-slate-500">
              <tr>
                <th className="p-3 w-10">Select</th>
                <th className="p-3">ID</th>
                <th className="p-3">ឈ្មោះ</th>
                <th className="p-3">Email</th>
                <th className="p-3">Role</th>
                <th className="p-3">Score</th>
              </tr>
            </thead>
            <tbody>
              {filteredRows.slice(0, 100).map((row) => (
                <GridRow
                  key={row.id}
                  item={row}
                  isSelected={selectedIds.has(row.id)}
                  onSelect={handleToggleSelect}
                />
              ))}
            </tbody>
          </table>
        </div>

      </div>
    </div>
  );
}
```

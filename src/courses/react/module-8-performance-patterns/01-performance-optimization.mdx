## 🛠️ 2. កូដអនុវត្តជាក់ស្តែង: `useMemo` & `useCallback`

```jsx
import { useState, useMemo, useCallback, memo } from 'react';

// 1. Child Component Memoized
const ExpensiveChart = memo(function ExpensiveChart({ data, onExport }) {
  console.log("ExpensiveChart rendered! 🎨");
  return (
    <div className="p-4 border rounded-xl bg-white shadow-sm mt-4">
      <p className="font-bold">Chart Points: {data.length}</p>
      <button onClick={onExport} className="text-xs bg-slate-900 text-white px-3 py-1 rounded mt-2">
        Export Data
      </button>
    </div>
  );
});

export default function PerformanceApp() {
  const [count, setCount] = useState(0);
  const [filterQuery, setFilterQuery] = useState('');

  // 2. useMemo: Cache Expensive Calculation
  const filteredData = useMemo(() => {
    console.log("Computing heavy data filter... ⏳");
    const rawData = Array.from({ length: 5000 }, (_, i) => `Item #${i + 1}`);
    return rawData.filter((item) => item.includes(filterQuery));
  }, [filterQuery]); // Re-calculate តែពេល filterQuery ដូរ

  // 3. useCallback: Cache Function Reference
  const handleExport = useCallback(() => {
    alert("Exported successfully!");
  }, []); // Stable reference មិនប្តូរតាម count state ឡើយ

  return (
    <div className="p-8 max-w-md mx-auto">
      <div className="flex gap-4 items-center mb-6">
        <button 
          onClick={() => setCount((c) => c + 1)}
          className="px-4 py-2 bg-blue-600 text-white rounded-lg font-bold"
        >
          Count: {count}
        </button>
        <span className="text-xs text-gray-500">(ការចុច Count មិនបង្កឱ្យ Chart re-render ឡើយ)</span>
      </div>

      <input
        type="text"
        value={filterQuery}
        onChange={(e) => setFilterQuery(e.target.value)}
        placeholder="ស្វែងរក Item..."
        className="w-full border px-4 py-2 rounded-xl"
      />

      <ExpensiveChart data={filteredData} onExport={handleExport} />
    </div>
  );
}
```

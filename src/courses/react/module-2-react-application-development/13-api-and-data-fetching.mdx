## 🌐 ការទាញយកទិន្នន័យជាមួយ `fetch`

នៅក្នុង JavaScript ទំនើប យើងប្រើប្រាស់អនុគមន៍ `fetch` (ជាប់មកជាមួយ Browser ស្រាប់) ដើម្បីធ្វើការហៅទិន្នន័យ (Network Request)។ ដោយសារតែវាស៊ីពេល (Asynchronous) យើងត្រូវប្រើវាជាមួយគូកនរបស់វាគឺ `async/await`។

## ⏳ ស្ថានភាពទាំង ៣ នៃការហៅ API (The Holy Trinity)

ការហៅ API ដែលមានលក្ខណៈអាជីព ត្រូវតែរួមបញ្ចូលការគ្រប់គ្រង State ចំនួន ៣: `ទិន្នន័យ (Data)`, `កំពុងរង់ចាំ (Loading)`, និង `កំហុស (Error)`។ យើងនឹងហៅវាចេញពីខាងក្នុង `useEffect`។

```jsx
import { useState, useEffect } from 'react';

function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    // យើងមិនអាចប្រើ async ផ្ទាល់លើ useEffect() បានទេ 
    // យើងត្រូវបង្កើតអនុគមន៍ថ្មីមួយនៅខាងក្នុង
    const fetchUsers = async () => {
      try {
        setLoading(true);
        // 1. ហៅទៅកាន់ API 
        const response = await fetch('https://jsonplaceholder.typicode.com/users');
        
        // 2. ឆែកមើលក្រែងលោមានកំហុស (ឧ. Error 404, 500)
        if (!response.ok) throw new Error('បរាជ័យក្នុងការទាញយកទិន្នន័យ');
        
        // 3. បម្លែងអក្សរទៅជា Object (JSON)
        const data = await response.json();
        
        // 4. រក្សាទុកក្នុង State
        setUsers(data);
      } catch (err) {
        setError(err.message); // បើមានបញ្ហា (ឧ. ដាច់ Internet) វានឹងហក់មកទីនេះ
      } finally {
        setLoading(false); // ឈប់ Loading (ទោះបីជោគជ័យឫបរាជ័យក៏ដោយ)
      }
    };

    fetchUsers();
  }, []); // ប្រអប់ទទេ មានន័យថារត់តែម្តងនៅពេល Load ទំព័រដំបូង

  // ការបង្ហាញរូបរាងទៅតាម State (Conditional Rendering)
  if (loading) return <h2>កំពុងស្វែងរក... ⏳</h2>;
  if (error) return <h2 style={{ color: 'red' }}>បញ្ហា: {error} ❌</h2>;

  return (
    <ul>
      {users.map(user => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}
```

## 📬 ការបញ្ជូនទិន្នន័យទៅ Server (POST Request)

នៅពេលអ្នក Submit Form (ឧទាហរណ៍ ការចុះឈ្មោះគណនីថ្មី) អ្នកមិនមែនគ្រាន់តែ "ទាញយក" (GET) នោះទេ ប៉ុន្តែអ្នកកំពុង "បញ្ជូន" (POST) ទិន្នន័យថ្មីទៅកាន់ Server។

អ្នកត្រូវកំណត់ `method`, ព្រមទាំងបន្ថែម `headers` និង `body` (ទិន្នន័យ)៖

```jsx
const registerUser = async (userData) => {
  try {
    const response = await fetch('https://api.example.com/register', {
      method: 'POST', // បញ្ជាក់ថាជាការបញ្ជូន
      headers: {
        'Content-Type': 'application/json', // ប្រាប់ Server ថាទិន្នន័យជា JSON
      },
      // បម្លែង Object ទៅជាអក្សរ JSON មុននឹងបញ្ជូន
      body: JSON.stringify(userData) 
    });

    if (response.ok) {
      alert("បង្កើតគណនីបានជោគជ័យ!");
    }
  } catch (error) {
    console.error("មានបញ្ហា:", error);
  }
};
```

> **⚡ គន្លឹះ:** បច្ចុប្បន្ន ក្រុមហ៊ុនធំៗតែងតែប្រើប្រាស់បណ្ណាល័យជំនួយដូចជា **Axios** ឫ **React Query** ដែលជួយកាត់បន្ថយកូដបានយ៉ាងច្រើន និងមានមុខងារ Cache យ៉ាងមានប្រសិទ្ធភាព!

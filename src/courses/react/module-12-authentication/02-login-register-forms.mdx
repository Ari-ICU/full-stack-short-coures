```html

</div>
```

## 💻 2. React Login Handler & API Submission

### 💻 Code សម្រាប់តែ handleSubmit Function:

```javascript
const [formData, setFormData] = useState({ email: '', password: '' });
const [isLoading, setIsLoading] = useState(false);
const [serverError, setServerError] = useState('');

const handleSubmit = async (e) => {
  e.preventDefault();
  setIsLoading(true);
  setServerError('');

  try {
    // 1. បាញ់ POST request ទៅ Server
    const response = await fetch('/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(formData),
    });

    const data = await response.json();

    // 2. ពិនិត្យ Response
    if (!response.ok) {
      throw new Error(data.message || 'អ៊ីម៉ែល ឬពាក្យសម្ងាត់មិនត្រឹមត្រូវ!');
    }

    // 3. Login ជោគជ័យ: រក្សាទុក Token និង User ទិន្នន័យ
    login(data.user, data.token);
    navigate('/dashboard', { replace: true });
  } catch (err) {
    setServerError(err.message);
  } finally {
    setIsLoading(false);
  }
};
```

### 🔍 ការពន្យល់មួយជួរៗ (Line-by-Line Explanation):

1. `e.preventDefault();`
   - ទប់ស្កាត់ការ Refresh ទំព័ររបស់ HTML Form។
2. `fetch('/api/auth/login', { method: 'POST', ... })`
   - បញ្ជូន Credentials ទៅកាន់ Backend ក្នុងទម្រង់ JSON body។
3. `if (!response.ok) throw new Error(...)`
   - ប្រសិនបើ Server ឆ្លើយតប status 400 ឬ 401 វានឹងបោះ Error ទៅ block `catch`។
4. `login(data.user, data.token)`
   - ហៅ Action ពី AuthContext ដើម្បី Save token ចូល LocalStorage និង Update Global State។
5. `navigate('/dashboard', { replace: true })`
   - នាំ User ទៅកាន់ Dashboard ភ្លាមៗ។

## 🚀 3. Complete Integrated LoginPage Component

```jsx
import { useState } from 'react';
import { useNavigate, Link, Navigate } from 'react-router-dom';
import { useAuth } from '../context/AuthContext';

export default function LoginPage() {
  const { isAuthenticated, login, isLoading: authLoading } = useAuth();
  const navigate = useNavigate();

  const [formData, setFormData] = useState({ email: '', password: '' });
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [serverError, setServerError] = useState('');

  // ប្រសិនបើ User បាន Login រួចហើយ មិនឱ្យមើល Login Page ទៀតទេ
  if (!authLoading && isAuthenticated) {
    return <Navigate to="/dashboard" replace />;
  }

  const handleSubmit = async (e) => {
    e.preventDefault();
    setIsSubmitting(true);
    setServerError('');

    try {
      // Demo authentication simulation
      await new Promise((r) => setTimeout(r, 1000));

      if (formData.email === 'admin@test.com' && formData.password === 'password123') {
        login({ id: '1', name: 'Admin User', email: formData.email, role: 'admin' }, 'mock-jwt-token');
        navigate('/dashboard', { replace: true });
      } else {
        throw new Error('Email: admin@test.com | Password: password123');
      }
    } catch (err) {
      setServerError(err.message);
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <div className="min-h-screen flex items-center justify-center bg-slate-50 dark:bg-slate-900 p-4">
      <div className="w-full max-w-sm bg-white dark:bg-slate-800 rounded-2xl shadow-lg p-6 border border-slate-100 dark:border-slate-700">
        <h2 className="text-2xl font-black text-center text-slate-800 dark:text-white mb-2">
          🔐 ចូលប្រើប្រាស់
        </h2>
        <p className="text-xs text-slate-400 text-center mb-6">បញ្ចូលគណនីរបស់អ្នកដើម្បីបន្ត</p>

        {serverError && (
          <div className="mb-4 p-3 bg-red-50 text-red-600 text-xs rounded-xl border border-red-200">
            ⚠️ {serverError}
          </div>
        )}

        <form onSubmit={handleSubmit} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold text-slate-600 dark:text-slate-300 mb-1">Email</label>
            <input
              type="email"
              required
              value={formData.email}
              onChange={(e) => setFormData((p) => ({ ...p, email: e.target.value }))}
              placeholder="admin@test.com"
              className="w-full px-3 py-2 border rounded-xl dark:bg-slate-700 dark:text-white text-sm"
            />
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-600 dark:text-slate-300 mb-1">Password</label>
            <input
              type="password"
              required
              value={formData.password}
              onChange={(e) => setFormData((p) => ({ ...p, password: e.target.value }))}
              placeholder="••••••••"
              className="w-full px-3 py-2 border rounded-xl dark:bg-slate-700 dark:text-white text-sm"
            />
          </div>

          <button
            type="submit"
            disabled={isSubmitting}
            className="w-full py-2.5 bg-blue-600 hover:bg-blue-700 disabled:opacity-50 text-white font-bold rounded-xl text-sm transition"
          >
            {isSubmitting ? 'កំពុងផ្ទៀងផ្ទាត់...' : 'ចូលគណនី →'}
          </button>
        </form>

        <p className="text-center text-xs text-slate-500 mt-4">
          មិនទាន់មានគណនី?{' '}
          <Link to="/register" className="text-blue-600 font-bold hover:underline">
            ចុះឈ្មោះទីនេះ
          </Link>
        </p>
      </div>
    </div>
  );
}
```

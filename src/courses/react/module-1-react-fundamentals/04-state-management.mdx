## 🧠 អ្វីទៅជា State?

នៅក្នុង Component មួយ ប្រសិនបើអ្នកចង់អោយទិន្នន័យមានការផ្លាស់ប្តូរ (ឧ. ពេលអ្នកប្រើប្រាស់វាយបញ្ចូលអក្សរ ឬចុចប៊ូតុង) ការប្រើប្រាស់អថេរធម្មតា (`let count = 0`) គឺមិនដំណើរការនោះទេ ព្រោះ React នឹងមិនដឹងថាទិន្នន័យនោះផ្លាស់ប្តូរ ហើយក៏មិនព្រមធ្វើបច្ចុប្បន្នភាពអេក្រង់ (Re-render) អោយអ្នកដែរ។

ដើម្បីដោះស្រាយបញ្ហានេះ React បានបង្កើត **State** ដែលប្រៀបដូចជា "អង្គចងចាំ" របស់ Component!

---

## 1. ការប្រើប្រាស់ `useState` Hook

នៅក្នុង React, ពាក្យថា **Hooks** គឺជាអនុគមន៍ពិសេសដែលអនុញ្ញាតអោយយើងទាញយកសមត្ថភាពរបស់ React មកប្រើ។ Hook ដែលពេញនិយមបំផុតគឺ `useState`។

```jsx
// 1. អ្នកត្រូវតែទាញយកវាពី react មកប្រើ
import { useState } from 'react';

export default function Counter() {
  // 2. ការប្រកាស State
  // 'count' គឺជាតម្លៃបច្ចុប្បន្ន
  // 'setCount' គឺជាកុងតាក់សម្រាប់ផ្លាស់ប្តូរតម្លៃនោះ
  // '0' គឺជាតម្លៃដំបូងគេបង្អស់ (Initial Value)
  const [count, setCount] = useState(0);

  const increment = () => {
    // 3. ការផ្លាស់ប្តូរតម្លៃ State
    setCount(count + 1); // ប្រាប់ React ថា "សុំបូក ១ ថែម ហើយគូរអេក្រង់ឡើងវិញផង!"
  };

  return (
    <div>
      <p>អ្នកបានចុចចំនួន {count} ដង</p>
      <button onClick={increment}>ចុចបូក</button>
    </div>
  );
}
```

> ⚠️ **ក្បួនសំខាន់:** អ្នកមិនអាចផ្លាស់ប្តូរ State ដោយផ្ទាល់បានទេ (ឧ. `count = count + 1` គឺខុស!)។ អ្នកត្រូវតែប្រើប្រាស់អនុគមន៍ដែលមកជាមួយវាជានិច្ច (`setCount()`)។

## 2. State និងប្រភេទនៃទិន្នន័យ

`useState` អាចរក្សាទុកទិន្នន័យបានគ្រប់ប្រភេទ!

```jsx
// ប្រភេទអក្សរ (String)
const [name, setName] = useState("សុខា");

// ប្រភេទបិទ/បើក (Boolean) សម្រាប់បង្កប់/បង្ហាញអ្វីមួយ
const [isOpen, setIsOpen] = useState(false);

// ប្រភេទ Array សម្រាប់រក្សាទុកទិន្នន័យច្រើន
const [items, setItems] = useState([]);

// ប្រភេទ Object (មានភាពស្មុគស្មាញបន្តិច)
const [user, setUser] = useState({ name: "រដ្ឋា", age: 20 });
```

## 3. ការ Update Object នៅក្នុង State

នៅពេលអ្នកធ្វើបច្ចុប្បន្នភាព (Update) Object ឬ Array អ្នកត្រូវតែបង្កើត Object/Array ថ្មីទាំងស្រុងជានិច្ច ជាជាងការកែប្រែដោយផ្ទាល់ (Mutation)។ យើងនិយមប្រើ `...` (Spread Operator) ដើម្បីចម្លងទិន្នន័យចាស់ រួចសឹមប្តូរតម្លៃថ្មី។

```jsx
const [user, setUser] = useState({ name: "រដ្ឋា", age: 20 });

const updateAge = () => {
  // ❌ ខុស: ការកែប្រែដោយផ្ទាល់ React នឹងមិន Re-render ទេ
  // user.age = 21; 
  
  // ✅ ត្រូវ: ចម្លងកូដចាស់មកដាក់សិន (...user) រួចប្តូរតែអាយុ
  setUser({ ...user, age: 21 });
};
```

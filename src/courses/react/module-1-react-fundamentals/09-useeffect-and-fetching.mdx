## ⚡ ស្គាល់ពី `useEffect` Hook

ជាទូទៅ កូដនៅក្នុង Component នឹងរត់តាំងពីលើដល់ក្រោមរាល់ពេលដែល State ផ្លាស់ប្តូរ (Re-render)។ ប៉ុន្តែពេលខ្លះ យើងចង់ដំណើរការកូដណាមួយតែម្តងគត់ (ឧ. ទាញយកទិន្នន័យពី API) ឬដំណើរការវាតែពេលមាន State ជាក់លាក់ណាមួយផ្លាស់ប្តូរប៉ុណ្ណោះ។ 

ដើម្បីធ្វើរឿងនេះបាន React បានបង្កើត Hook មួយឈ្មោះថា **`useEffect`** (Side Effects)។

---

## 1. របៀបប្រើប្រាស់ `useEffect`

`useEffect` ទទួលយកប៉ារ៉ាម៉ែត្រពីរ៖
1. **Callback Function**: កូដដែលអ្នកចង់អោយដំណើរការ។
2. **Dependency Array (ជម្រើស)**: បញ្ជីនៃអថេរឬ State ដែលកំណត់ថាតើ Function ខាងលើគួរដំណើរការឬទេ។

**ទម្រង់ទាំង ៣ របស់វា៖**

```jsx
import { useEffect, useState } from 'react';

export default function EffectExample() {
  const [count, setCount] = useState(0);

  // ១. គ្មាន Dependency Array: រត់រាល់ពេល Re-render ទាំងអស់! (កម្រប្រើណាស់)
  useEffect(() => {
    console.log("Component ដំណើរការ (Re-rendered)!");
  });

  // ២. មាន Array ទទេ []: រត់តែម្តងគត់នៅពេលបើកទំព័រដំបូង! (ពេញនិយមបំផុត)
  useEffect(() => {
    console.log("សួស្តី! ខ្ញុំរត់តែម្តងគត់ពេលទំព័របើក។");
  }, []);

  // ៣. មានអថេរក្នុង Array [count]: រត់តែពេលអថេរ count ផ្លាស់ប្តូរប៉ុណ្ណោះ!
  useEffect(() => {
    console.log("ចំនួនត្រូវបានផ្លាស់ប្តូរទៅជា: " + count);
  }, [count]);

  return <button onClick={() => setCount(count + 1)}>ចុចទីនេះ</button>;
}
```

## 2. ការទាញយកទិន្នន័យពី API (Data Fetching)

ករណីប្រើប្រាស់ `useEffect` ដ៏ពេញនិយមបំផុត គឺការហៅ API ដើម្បីយកទិន្នន័យមកបង្ហាញនៅលើអេក្រង់នៅពេលអ្នកប្រើប្រាស់ចូលមកដល់ទំព័រនោះភ្លាម។ យើងត្រូវផ្សំវាចូលគ្នាជាមួយនឹង `useState`។

```jsx
export default function UserList() {
  // បង្កើត State សម្រាប់ផ្ទុកទិន្នន័យដែលយកបានពី API
  const [users, setUsers] = useState([]);
  // State សម្រាប់កំណត់ថាវាបញ្ឈប់ការ Loading ឬនៅ
  const [loading, setLoading] = useState(true);

  // ដំណើរការការទាញយកទិន្នន័យតែម្តងគត់ ពេល Component បង្ហាញខ្លួន
  useEffect(() => {
    // ដោយសារយើងមិនអាចប្រើប្រាស់ async ផ្ទាល់លើ useEffect យើងត្រូវបង្កើត function ខាងក្នុង
    const fetchUsers = async () => {
      try {
        const response = await fetch("https://jsonplaceholder.typicode.com/users");
        const data = await response.json();
        setUsers(data); // ទុកទិន្នន័យក្នុង State
      } catch (error) {
        console.error("មានបញ្ហា:", error);
      } finally {
        setLoading(false); // បញ្ឈប់ Loading
      }
    };

    fetchUsers(); // ហៅ function នោះ
  }, []); // Array ទទេ! សំខាន់ខ្លាំងណាស់ ជៀសវាងវាហៅ API អត់ឈប់ (Infinite loop)!

  if (loading) return <p>កំពុងផ្ទុកទិន្នន័យ...</p>;

  return (
    <ul>
      {users.map(user => (
        <li key={user.id}>{user.name} - {user.email}</li>
      ))}
    </ul>
  );
}
```

## 3. ការសម្អាត (Cleanup Function)

ពេលខ្លះ ផលប៉ះពាល់ (Effects) របស់អ្នកតម្រូវអោយមានការសម្អាតនៅពេល Component ត្រូវបានបិទចោល (Unmount) ឧទាហរណ៍ដូចជាការបញ្ឈប់ Timer ជាដើម។ អ្នកគ្រាន់តែ `return` function មួយត្រលប់មកវិញនៅខាងក្នុង `useEffect`។

```jsx
useEffect(() => {
  const timer = setInterval(() => {
    console.log("កំពុងរាប់ម៉ោង...");
  }, 1000);

  // Cleanup Function នឹងដើរនៅពេល Component នេះត្រូវបានលុបចោលពីអេក្រង់
  return () => {
    clearInterval(timer);
    console.log("បញ្ឈប់ការរាប់ម៉ោង!");
  };
}, []);
```

## 📝 ការគ្រប់គ្រងទម្រង់ (Forms) នៅក្នុង React

នៅក្នុងមេរៀនមុន អ្នកបានរៀនពីការប្រើ `onChange` ដើម្បីចាប់យកអក្សរពី Input មួយ។ បច្ចេកទេសនោះត្រូវបានគេហៅថា **Controlled Component**។ មានន័យថា React គឺជាអ្នកគ្រប់គ្រងទិន្នន័យដែលបង្ហាញនៅក្នុង Input នោះទាំងស្រុង តាមរយៈ State។

---

## 1. របៀបសរសេរ Controlled Component

ដើម្បីអោយ Input មួយក្លាយជា Controlled Component វាត្រូវតែមានលក្ខខណ្ឌ ២ យ៉ាង៖
1. មាន `value={state}` ដើម្បីអោយ React កំណត់តម្លៃអោយវា។
2. មាន `onChange` ដើម្បីអោយយើងចាប់យកអក្សរថ្មីទៅដាក់ក្នុង State វិញ។

```jsx
import { useState } from 'react';

export default function SimpleForm() {
  const [name, setName] = useState("");

  return (
    <form>
      {/* នេះគឺជា Controlled Component */}
      <input 
        type="text" 
        value={name} 
        onChange={(e) => setName(e.target.value)} 
      />
      <p>ឈ្មោះរបស់អ្នកគឺ: {name}</p>
    </form>
  );
}
```

## 2. ការគ្រប់គ្រង Input ច្រើន (Multiple Inputs)

នៅពេលដែលទម្រង់របស់អ្នកមាន Input ច្រើន (ឧ. ឈ្មោះ, អុីមែល, លេខសម្ងាត់) ការបង្កើត State មួយសម្រាប់ Input មួយគឺជារឿងហត់នឿយណាស់! អ្នកអាចប្រើ **Object State** តែមួយដើម្បីគ្រប់គ្រងពួកវាទាំងអស់បាន។ 

**គន្លឹះសំខាន់:** អ្នកត្រូវដាក់គុណលក្ខណៈ `name=""` អោយ Input នីមួយៗអោយដូចទៅនឹងឈ្មោះ key នៅក្នុង State។

```jsx
export default function RegisterForm() {
  // បង្កើត State ជា Object តែមួយ
  const [formData, setFormData] = useState({
    username: "",
    email: "",
    password: ""
  });

  const handleChange = (e) => {
    // ចាប់យកឈ្មោះ (name) និងតម្លៃ (value) របស់ Input ដែលគេកំពុងវាយ
    const { name, value } = e.target;
    
    // ចម្លងទិន្នន័យចាស់ រួចកែប្រែតែផ្នែកដែលផ្លាស់ប្តូរ
    setFormData({
      ...formData,
      [name]: value // [name] នឹងប្រែទៅជា 'username', 'email' ជាដើម
    });
  };

  return (
    <form>
      <input name="username" value={formData.username} onChange={handleChange} placeholder="ឈ្មោះ" />
      <input name="email" value={formData.email} onChange={handleChange} placeholder="អុីមែល" />
      <input name="password" type="password" value={formData.password} onChange={handleChange} placeholder="លេខសម្ងាត់" />
    </form>
  );
}
```

## 3. ការត្រួតពិនិត្យទិន្នន័យ (Validation)

មុនពេលផ្ញើទិន្នន័យទៅកាន់ Server យើងគួរតែពិនិត្យមើលសិនថាតើអ្នកប្រើប្រាស់បានបំពេញត្រឹមត្រូវឬទេ។ យើងអាចបង្កើត State មួយទៀតសម្រាប់ផ្ទុកសារកំហុស (Errors)។

```jsx
export default function FormWithValidation() {
  const [email, setEmail] = useState("");
  const [error, setError] = useState("");

  const handleSubmit = (e) => {
    e.preventDefault();
    
    // ការត្រួតពិនិត្យ (Validation)
    if (email === "") {
      setError("សូមបំពេញអុីមែលរបស់អ្នក!");
      return; // បញ្ឈប់កូដនៅត្រឹមនេះ
    }
    
    if (!email.includes("@")) {
      setError("អុីមែលមិនត្រឹមត្រូវទេ!");
      return;
    }

    // បើត្រឹមត្រូវ
    setError("");
    alert("ទម្រង់ត្រឹមត្រូវ! កំពុងផ្ញើទិន្នន័យ...");
  };

  return (
    <form onSubmit={handleSubmit}>
      <div>
        <input 
          type="text" 
          value={email} 
          onChange={(e) => setEmail(e.target.value)} 
          style={{ borderColor: error ? 'red' : 'black' }}
        />
        {/* បង្ហាញកំហុសបើមាន */}
        {error && <p style={{ color: 'red' }}>{error}</p>}
      </div>
      <button type="submit">Submit</button>
    </form>
  );
}
```

## ⚖️ ការបង្ហាញតាមលក្ខខណ្ឌ (Conditional Rendering)

នៅក្នុង React ការបង្ហាញមាតិកាខុសៗគ្នាទៅតាមស្ថានភាពជាក់ស្តែង (ឧទាហរណ៍៖ បង្ហាញសញ្ញាកំពុងផ្ទុក (Loading spinner) ពេលទិន្នន័យមិនទាន់មកដល់) គឺជារឿងដែលយើងប្រើប្រាស់ស្ទើរតែរាល់ថ្ងៃ។ 

អ្នកមិនអាចសរសេរ `if...else` ធម្មតានៅខាងក្នុងវង់ក្រចករបស់ JSX `{}` បានទេ។ ដូច្នេះយើងត្រូវប្រើប្រាស់បច្ចេកទេស ៣ យ៉ាងដូចខាងក្រោម៖

---

## 1. ការប្រើ Ternary Operator (`? :`)

នេះគឺជាវិធីដែលគេនិយមប្រើបំផុត! វាយកមកជំនួស `if...else` ដើម្បីអោយយើងអាចសរសេរវាចូលទៅក្នុង JSX បាន។

**ទម្រង់:** `{ លក្ខខណ្ឌ ? (បង្ហាញកូដនេះបើពិត) : (បង្ហាញកូដនេះបើមិនពិត) }`

```jsx
import { useState } from 'react';

export default function App() {
  const [isLogin, setIsLogin] = useState(false);

  return (
    <div>
      <h1>ប្រព័ន្ធគ្រប់គ្រងសិស្ស</h1>
      
      {/* បើ isLogin = true បង្ហាញស្វាគមន៍, បើ false បង្ហាញប៊ូតុង Login */}
      {isLogin ? (
        <p>សួស្តី! សូមស្វាគមន៍មកកាន់គណនីរបស់អ្នក។</p>
      ) : (
        <button onClick={() => setIsLogin(true)}>ចូលគណនី (Login)</button>
      )}
      
    </div>
  );
}
```

## 2. ការប្រើ Logical AND (`&&`)

ចុះបើអ្នកចង់បង្ហាញអ្វីមួយតែពេលលក្ខខណ្ឌពិត ហើយមិនចង់បង្ហាញអ្វីសោះបើវាមិនពិត? អ្នកមិនចាំបាច់ប្រើ `? ... : null` ទេ អ្នកអាចប្រើ `&&` បានយ៉ាងងាយ។

**ទម្រង់:** `{ លក្ខខណ្ឌ && (បង្ហាញកូដនេះបើពិត) }`

```jsx
export default function MessageCount({ unreadMessages }) {
  return (
    <div>
      <h2>ប្រអប់សាររបស់អ្នក</h2>
      
      {/* បង្ហាញសារជូនដំណឹងនេះ តែក្នុងករណីដែលមានសារថ្មី > 0 ប៉ុណ្ណោះ! */}
      {unreadMessages > 0 && (
        <div className="alert alert-warning">
          អ្នកមានសារថ្មីចំនួន {unreadMessages}!
        </div>
      )}
      
    </div>
  );
}
```

## 3. Early Return (ការបញ្ឈប់ពីក្បាលទី)

ប្រសិនបើអ្នកចង់លាក់ Component ទាំងមូលតែម្តង (ឧទាហរណ៍ មិនទាន់មានទិន្នន័យ) អ្នកអាចប្រើប្រាស់ `if` ធម្មតានៅពីលើ `return` ធំបាន! ការធ្វើបែបនេះគេហៅថា Early Return ដែលវាជួយកាត់បន្ថយភាពស្មុគស្មាញនៃកូដ (Nested code)។

```jsx
export default function UserProfile({ user }) {
  
  // បើមិនទាន់មានព័ត៌មានអ្នកប្រើប្រាស់ទេ បង្ហាញ Loading ហើយមិនបាច់អានកូដខាងក្រោមបន្តទៀតទេ
  if (!user) {
    return <p>កំពុងទាញយកទិន្នន័យ (Loading...)</p>;
  }

  // បើមាន user វានឹងដំណើរការកូដខាងក្រោមនេះជាធម្មតា
  return (
    <div>
      <img src={user.avatar} alt="Profile" />
      <h3>{user.name}</h3>
    </div>
  );
}
```

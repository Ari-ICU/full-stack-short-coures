## 📁 ការបញ្ចូលរូបភាព និងឯកសារ (File Upload)

នៅក្នុង React ការចាប់យកឯកសារពីអ្នកប្រើប្រាស់គឺមានលក្ខណៈខុសប្លែកពីការចាប់យកអក្សរធម្មតាបន្តិច ព្រោះឯកសារ (File) គឺជាប្រភេទមេឌៀ (Binary Data) មិនមែនជាអក្សរ (String) ទេ។ យើងមិនអាចប្រើប្រាស់ `e.target.value` បានទេ!

---

## 1. របៀបចាប់យកឯកសារ (The Basics)

យើងត្រូវប្រើប្រាស់ `e.target.files` ដើម្បីចាប់យកឯកសារដែលគេរើស។ `files` គឺជាបញ្ជីមួយ ព្រោះអ្នកប្រើប្រាស់អាចជ្រើសរើសឯកសារច្រើនក្នុងពេលតែមួយ (ប្រសិនបើមានដាក់គុណលក្ខណៈ `multiple`)។

```jsx
import { useState } from 'react';

export default function FileUploader() {
  const [selectedFile, setSelectedFile] = useState(null);

  const handleFileChange = (e) => {
    // ចាប់យកឯកសារទី១ ដែលគេបានរើស
    const file = e.target.files[0];
    
    if (file) {
      setSelectedFile(file); // រក្សាទុកក្នុង State
      console.log("ឈ្មោះឯកសារ:", file.name);
      console.log("ទំហំ:", file.size, "bytes");
    }
  };

  return (
    <div>
      <h3>សូមជ្រើសរើសរូបភាពរបស់អ្នក</h3>
      <input type="file" onChange={handleFileChange} accept="image/*" />
      
      {selectedFile && (
        <p style={{ color: 'green' }}>
          អ្នកបានជ្រើសរើស: {selectedFile.name}
        </p>
      )}
    </div>
  );
}
```

## 2. ការបង្ហាញរូបភាពមុនពេលបញ្ជូន (Image Preview) 🖼️

ជារឿយៗ នៅពេលដែលយើងប្តូរ Profile Picture យើងចង់ឃើញរូបភាពនោះភ្លាមៗនៅលើអេក្រង់ មុនពេលយើងចុចប៊ូតុង Save។ តើយើងអាចបង្ហាញរូបភាពចេញពីកុំព្យូទ័រដោយមិនទាន់មាន URL ផ្លូវការយ៉ាងដូចម្តេច?

JavaScript មានអនុគមន៍មួយឈ្មោះថា `URL.createObjectURL()` ដែលអាចបង្កើត URL ក្លែងក្លាយ (បណ្តោះអាសន្ន) សំរាប់ឯកសារក្នុងកុំព្យូទ័រយើងបាន!

```jsx
import { useState } from 'react';

export default function ImagePreviewer() {
  const [selectedImage, setSelectedImage] = useState(null);

  const handleImageChange = (e) => {
    const file = e.target.files[0];
    if (file) {
      // បង្កើត URL បណ្តោះអាសន្នដើម្បីយកទៅដាក់ក្នុង <img src="..." />
      const imageUrl = URL.createObjectURL(file);
      setSelectedImage(imageUrl);
    }
  };

  return (
    <div>
      <h3>បញ្ចូលរូបភាព Profile</h3>
      <input type="file" onChange={handleImageChange} accept="image/png, image/jpeg" />
      
      <br /><br />
      
      {/* បង្ហាញរូបភាពបើសិនជាមាន */}
      {selectedImage && (
        <div>
          <p>រូបភាពមើលមុន (Preview):</p>
          <img 
            src={selectedImage} 
            alt="Preview" 
            style={{ width: "200px", height: "200px", borderRadius: "50%", objectFit: "cover" }} 
          />
        </div>
      )}
    </div>
  );
}
```

## 3. ការបញ្ជូនឯកសារទៅកាន់ Server (API)

នៅពេលបញ្ជូនឯកសារទៅកាន់ Backend អ្នកមិនអាចប្រើ JSON ធម្មតាបានទេ។ អ្នកត្រូវប្រើប្រាស់ `FormData`។

```jsx
const handleUpload = async () => {
  if (!selectedFile) return;

  // បង្កើតកញ្ចប់ទិន្នន័យ FormData
  const formData = new FormData();
  formData.append("profilePicture", selectedFile);
  formData.append("username", "Ratha"); // អាចថែមអក្សរធម្មតាចូលបានដែរ

  // ផ្ញើទៅ API
  try {
    await fetch("https://api.example.com/upload", {
      method: "POST",
      body: formData // កត់សម្គាល់ថា យើងមិនបាច់ប្រាប់ Headers ទេ FormData នឹងរៀបចំដោយខ្លួនឯង
    });
    alert("បញ្ជូនជោគជ័យ!");
  } catch (err) {
    console.error(err);
  }
};
```

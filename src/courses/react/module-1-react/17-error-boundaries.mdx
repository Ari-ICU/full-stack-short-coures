## 💥 បញ្ហានៃការគាំង (Crashes)

នៅក្នុង React ប្រសិនបើមាន Component ណាមួយមានកំហុស (Error) អំឡុងពេលវាកំពុងគូរនៅលើអេក្រង់ (Render) នោះវានឹងធ្វើអោយកម្មវិធី React ទាំងមូលគាំង ហើយអេក្រង់របស់អ្នកនឹងប្រែជាពណ៌សទទេ (White screen of death)។ 

នេះជារឿងដ៏អាក្រក់បំផុតសម្រាប់អ្នកប្រើប្រាស់! តើយើងអាចការពារវាបានដោយរបៀបណា?

---

## 1. អ្វីទៅជា Error Boundary?

**Error Boundary** គឺជា Component ពិសេសមួយដែលដើរតួជាសំណាញ់ការពារ។ ប្រសិនបើ Component កូនណាមួយរបស់វាមានកំហុស វាអាច "ចាប់យក" កំហុសនោះបាន ហើយបង្ហាញផ្ទាំងសារសុំទោស (Fallback UI) ជាជាងទុកអោយគេហទំព័រទាំងមូលគាំង។

> ⚠️ **ចំណាំ:** បច្ចុប្បន្ន Error Boundary អាចសរសេរបានតែជាទម្រង់ **Class Component** ប៉ុណ្ណោះ (មិនទាន់អាចសរសេរជាទម្រង់ Functional Component ជាមួយ Hooks បាននៅឡើយទេ)។ ទោះជាយ៉ាងណាក៏ដោយ យើងសរសេរវាតែម្តងគត់ ហើយយកវាទៅរុំពេញគម្រោងតែម្តង។

## 2. ការបង្កើត Error Boundary

**ឯកសារ `ErrorBoundary.jsx`:**
```jsx
import React from 'react';

export class ErrorBoundary extends React.Component {
  constructor(props) {
    super(props);
    // បង្កើត State សម្រាប់កត់ចំណាំថាមានកំហុសឬអត់
    this.state = { hasError: false };
  }

  // អនុគមន៍នេះនឹងរត់ដោយស្វ័យប្រវត្តិនៅពេលមាន Component កូនណាមាន Error
  static getDerivedStateFromError(error) {
    // ផ្លាស់ប្តូរ State អោយវាដឹងថាមាន Error
    return { hasError: true };
  }

  componentDidCatch(error, errorInfo) {
    // អ្នកអាចយក Error នេះទៅរក្សាទុកក្នុង Server (Log) ក៏បាន
    console.error("មានកំហុសមួយត្រូវបានចាប់បាន:", error, errorInfo);
  }

  render() {
    // ប្រសិនបើមាន Error បង្ហាញ UI នេះ!
    if (this.state.hasError) {
      return (
        <div style={{ padding: 20, background: '#fee', color: 'red', border: '1px solid red' }}>
          <h2>អូស! មានបញ្ហាអ្វីម្យ៉ាងកើតឡើងហើយ។</h2>
          <p>សូមធ្វើការ Refresh ទំព័រឡើងវិញ។</p>
        </div>
      );
    }

    // បើគ្មាន Error ទេ បង្ហាញ Component កូនធម្មតា
    return this.props.children; 
  }
}
```

## 3. ការប្រើប្រាស់ Error Boundary

ឥឡូវនេះ យើងយកសំណាញ់នេះទៅរុំ Component ណាមួយដែលយើងសង្ស័យថាអាចនឹងមានបញ្ហា ឬយកទៅរុំ Component ធំ (Root) តែម្តងក៏បាន។

```jsx
import { ErrorBoundary } from './ErrorBoundary';
import BuggyComponent from './BuggyComponent'; // Component ដែលមានកំហុស

export default function App() {
  return (
    <div>
      <h1>កម្មវិធីលក់ទំនិញ</h1>
      
      {/* ទោះបីជាមាន Error នៅក្នុង BuggyComponent ក៏វាមិនធ្វើអោយទំព័រទាំងមូលគាំងដែរ! */}
      <ErrorBoundary>
        <BuggyComponent />
      </ErrorBoundary>
      
      <p>មាតិកាផ្សេងៗទៀតនៅខាងក្រោមនៅតែដំណើរការជាធម្មតា!</p>
    </div>
  );
}
```

**សន្និដ្ឋាន:** Error Boundaries គឺជាខ្សែការពារចុងក្រោយដ៏សំខាន់សម្រាប់ធានាថាកម្មវិធីរបស់អ្នកមិនគាំងទាំងស្រុងនៅចំពោះមុខអ្នកប្រើប្រាស់នៅពេលដែលមានអ្វីមួយខុសប្រក្រតី។

## 🖱️ ការចាប់យកព្រឹត្តិការណ៍ (Event Handling)

គេហទំព័រដែលរស់រវើក គឺតែងតែមានការចុច និងការបញ្ជារបស់អ្នកប្រើប្រាស់ជានិច្ច។ នៅក្នុង React, ការចាប់យកព្រឹត្តិការណ៍ (Events) គឺមានលក្ខណៈស្រដៀងទៅនឹង HTML ធម្មតាដែរ តែមានច្បាប់សំខាន់ៗមួយចំនួនត្រូវចាំ។

**ច្បាប់ក្នុងការសរសេរ Events ក្នុង React:**
1. ឈ្មោះ Event ត្រូវសរសេរជាទម្រង់ **camelCase** (ឧ. ពី `onclick` ទៅជា `onClick`, ពី `onchange` ទៅជា `onChange`)។
2. អ្នកត្រូវបញ្ជូន Function ចូលទៅក្នុងវង់ក្រចក `{}` (មិនមែនជាអក្សរ `""` ទេ)។

---

## 1. ព្រឹត្តិការណ៍ចុចប៊ូតុង (`onClick`)

នៅពេលអ្នកចង់អោយកូដអ្វីមួយដើរពេលគេចុចប៊ូតុង អ្នកគ្រាន់តែបង្កើត Function មួយ ហើយបោះវាទៅអោយ `onClick`។

```jsx
export default function ButtonClick() {
  
  // បង្កើត Function សំរាប់ធ្វើអ្វីមួយ
  const handleClick = () => {
    alert("អ្នកបានចុចប៊ូតុង!");
  };

  return (
    // ❌ ខុស: onClick={handleClick()} <- វាដើរមុនពេលយើងចុចទៅទៀតព្រោះមាន ()
    // ✅ ត្រូវ: គ្រាន់តែប្រាប់ឈ្មោះ Function
    <button onClick={handleClick}>ចុចមក</button>
  );
}
```

> **ប្រសិនបើអ្នកចង់បោះ Argument ចូលទៅក្នុង Function នោះ តើត្រូវធ្វើយ៉ាងណា?**
អ្នកត្រូវរុំវាដោយប្រើប្រាស់ Arrow function ខ្លីមួយសិន៖
`<button onClick={() => deleteUser(5)}>លុបគណនី</button>`

## 2. ព្រឹត្តិការណ៍វាយអក្សរ (`onChange`)

នៅពេលអ្នកចង់ដឹងថាអ្នកប្រើប្រាស់កំពុងវាយអ្វីចូលទៅក្នុងប្រអប់ `<input>` អ្នកត្រូវប្រើ `onChange`។ រាល់ពេលមានព្រឹត្តិការណ៍កើតឡើង React នឹងបញ្ជូន Object តូចមួយឈ្មោះថា `event` (ឬ `e`) មកអោយយើង ដែលផ្ទុកព័ត៌មានជាច្រើន។

យើងអាចទាញយកអក្សរដែលគេវាយបញ្ចូលតាមរយៈ `e.target.value`។

```jsx
import { useState } from 'react';

export default function SearchBox() {
  const [text, setText] = useState("");

  const handleChange = (e) => {
    // ចាប់យកអក្សរដែលគេកំពុងវាយ ហើយបញ្ជូនវាទៅទុកក្នុង State
    setText(e.target.value);
  };

  return (
    <div>
      <input 
        type="text" 
        placeholder="ស្វែងរក..." 
        onChange={handleChange} 
      />
      <p>អ្នកកំពុងវាយពាក្យ: {text}</p>
    </div>
  );
}
```

## 3. ព្រឹត្តិការណ៍បញ្ជូនទម្រង់ (`onSubmit`)

ជាធម្មតា នៅពេលអ្នកចុចប៊ូតុង Submit នៅក្នុង `<form>` Browser នឹង Refresh ទំព័រទាំងមូលដោយស្វ័យប្រវត្តិ។ នៅក្នុង React យើងចង់គ្រប់គ្រងវាដោយខ្លួនឯង ដូច្នេះយើងត្រូវបញ្ឈប់ការ Refresh នោះដោយប្រើ `e.preventDefault()`។

```jsx
export default function LoginForm() {
  const handleSubmit = (e) => {
    e.preventDefault(); // បញ្ឈប់ការ Refresh ទំព័រ!
    alert("ទម្រង់ត្រូវបានបញ្ជូន!");
    // នៅទីនេះ អ្នកអាចសរសេរកូដដើម្បីផ្ញើទិន្នន័យទៅកាន់ Server (API)
  };

  return (
    <form onSubmit={handleSubmit}>
      <input type="text" placeholder="ឈ្មោះ" />
      <input type="password" placeholder="លេខសម្ងាត់" />
      <button type="submit">ចូលគណនី</button>
    </form>
  );
}
```

## 🎨 1. Vanilla CSS vs Tailwind CSS

### 🌐 Normal HTML & Traditional CSS:

```html
<!-- HTML Card -->
<div class="user-card">
  <img src="/avatar.jpg" class="avatar" />
  <div class="details">
    <h4 class="name">Sok Dara</h4>
    <p class="role">Frontend Developer</p>
  </div>
</div>

<style>
  .user-card {
    display: flex;
    align-items: center;
    gap: 16px;
    padding: 16px;
    background: #ffffff;
    border-radius: 16px;
    box-shadow: 0 4px 6px -1px rgba(0, 0, 0, 0.1);
  }
  .avatar {
    width: 48px;
    height: 48px;
    border-radius: 9999px;
  }
</style>
```

### 💻 React Code ជាមួយ Tailwind CSS:

```jsx
export function UserProfileCard({ name, role, avatar }) {
  return (
    <div className="flex items-center gap-4 p-4 bg-white dark:bg-slate-800 rounded-2xl shadow-md border border-slate-100 dark:border-slate-700">
      <img
        src={avatar}
        alt={name}
        className="w-12 h-12 rounded-full object-cover shadow-sm"
      />
      <div>
        <h4 className="font-bold text-slate-800 dark:text-white text-sm">{name}</h4>
        <p className="text-xs text-slate-500 dark:text-slate-400">{role}</p>
      </div>
    </div>
  );
}
```

### 🔍 ការពន្យល់មួយជួរៗ (Line-by-Line Explanation):

1. `flex items-center gap-4`
   - `display: flex; align-items: center; gap: 1rem;` (រៀបចំ layout ផ្ដេក និងតម្រឹមចំកណ្តាល)។
2. `bg-white dark:bg-slate-800`
   - គាំទ្រ Dark Mode ដោយស្វ័យប្រវត្តិ — ផ្ទៃសពេល Light Mode និងផ្ទៃប្រផេះក្រម៉ៅពេល Dark Mode។
3. `rounded-2xl shadow-md`
   - `border-radius: 1rem; box-shadow: ...` បង្កើតគែមកោង និងស្រមោលទំនើប។
4. `w-12 h-12 rounded-full`
   - កំណត់ទំហំ 48px × 48px និងរាងមូល ១០០% សម្រាប់រូបភាព Profile។

## 🚀 2. Complete Responsive Grid Layout

```jsx
const MEMBERS = [
  { id: 1, name: 'Sok Dara', role: 'React Lead', avatar: 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=100' },
  { id: 2, name: 'Chan Bopha', role: 'UI/UX Designer', avatar: 'https://images.unsplash.com/photo-1517841905240-472988babdf9?w=100' },
];

export default function TeamGrid() {
  return (
    <div className="max-w-xl mx-auto p-6 space-y-4">
      <h3 className="text-xl font-bold text-slate-800 dark:text-white">
        👥 Team Members (Tailwind Grid)
      </h3>

      {/* Responsive Grid: 1 Col លើ Mobile | 2 Cols លើ Tablet/Desktop */}
      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
        {MEMBERS.map((member) => (
          <div
            key={member.id}
            className="flex items-center gap-3 p-4 bg-white dark:bg-slate-800 rounded-2xl shadow border hover:shadow-lg transition cursor-pointer"
          >
            <img src={member.avatar} alt={member.name} className="w-10 h-10 rounded-full object-cover" />
            <div>
              <h4 className="font-bold text-xs text-slate-800 dark:text-white">{member.name}</h4>
              <p className="text-[11px] text-blue-500 font-semibold">{member.role}</p>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
```

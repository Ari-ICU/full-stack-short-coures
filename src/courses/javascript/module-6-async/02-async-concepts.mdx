## ⏳ អ្វីទៅជា Asynchronous JavaScript?

តាមធម្មជាតិ JavaScript គឺ **Synchronous** ពោលគឺវាធ្វើការម្តងមួយបន្ទាត់ៗពីលើចុះក្រោម។ បើបន្ទាត់ទី ១ ដើរមិនទាន់ចប់ទេ បន្ទាត់ទី ២ ត្រូវតែឈរចាំ។

ចុះបើបន្ទាត់ទី ១ ជាការទាញយកទិន្នន័យពី Server ដែលត្រូវចំណាយពេល ៥ វិនាទី? តើគេហទំព័រទាំងមូលត្រូវគាំង ៥ វិនាទីដែរឬ? ចម្លើយគឺ **អត់ទេ!** នេះហើយជាមូលហេតុដែលយើងមាន **Asynchronous**។

## 🤝 1. អ្វីទៅជា Promise?

ដើម្បីគ្រប់គ្រង Asynchronous កូដ អោយមានរបៀបរៀបរយ JavaScript បង្កើតនូវ **Promise** (ការសន្យា)។

*   **Pending:** កំពុងឆុងកាហ្វេ (មិនទាន់បានលទ្ធផល)
*   **Fulfilled:** ឆុងឆ្អិន អ្នកទទួលបានកាហ្វេ (ជោគជ័យ)
*   **Rejected:** អស់កាហ្វេ (បរាជ័យ/Error)

const status = document.getElementById("promiseStatus");

function makePromise(shouldResolve) {
  status.className = "";
  status.textContent = "⏳ Pending — Promise is pending...";

  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (shouldResolve) resolve({ name: "Sok", age: 25 });
      else reject(new Error("Network Error"));
    }, 1500);
  });
}

document.getElementById("resolveBtn").onclick = () => {
  makePromise(true)
    .then(data => {
      status.className = "fulfilled";
      status.textContent = "✅ Fulfilled — data: " + JSON.stringify(data);
    });
};

document.getElementById("rejectBtn").onclick = () => {
  makePromise(false)
    .catch(err => {
      status.className = "rejected";
      status.textContent = "❌ Rejected — " + err.message;
    });
};

document.getElementById("resetBtn").onclick = () => {
  status.className = "";
  status.textContent = "⏳ Pending...";
};

## 🚀 2. ការប្រើប្រាស់ `async` / `await`

`async/await` ធ្វើអោយកូដ Asynchronous មើលទៅស្រួលអានដូចកូដ Synchronous ធម្មតា!

1. ដាក់ `async` នៅពីមុខ Function
2. ដាក់ `await` នៅពីមុខ Promise — ផ្អាករង់ចាំតែ function ខាងក្នុង
3. ប្រើ `try/catch` ចាប់ error

async function fetchUserData() {
  const spinner = document.getElementById("loadingSpinner");
  const userData = document.getElementById("userData");

  spinner.style.display = "block";
  userData.textContent = "";
  userData.style.background = "";

  try {
    // simulate fetch delay
    const data = await new Promise(resolve =>
      setTimeout(() => resolve({ name: "Sok", age: 25, role: "Admin" }), 2000)
    );

    spinner.style.display = "none";
    userData.textContent = "✅ " + JSON.stringify(data);
    userData.style.background = "#f0fdf4";
    userData.style.border = "2px solid #86efac";
    userData.style.color = "#166534";

  } catch (error) {
    spinner.style.display = "none";
    userData.textContent = "❌ Error: " + error.message;
    userData.style.background = "#fff1f2";
    userData.style.color = "#dc2626";
  }
}

document.getElementById("fetchBtn").onclick = () => {
  const outside = document.getElementById("outsideCode");
  outside.textContent = "";
  fetchUserData();
  // This runs immediately — doesn't wait for fetchUserData
  outside.textContent = "▶ ខ្ញុំជាកូដខាងក្រៅ ខ្ញុំរត់មុន await!";
};

> 💡 **សរុបមក៖** `async/await` គឺជាវិធីដ៏ងាយស្រួលបំផុត និងជាស្តង់ដារបច្ចុប្បន្នក្នុងការទាញយកទិន្នន័យ (Fetch Data) ពី Server ដែលយើងនឹងអនុវត្តផ្ទាល់នៅក្នុងមេរៀនបន្ទាប់!

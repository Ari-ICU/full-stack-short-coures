## 🛑 ហេតុអ្វីត្រូវមាន Error Handling?

នៅក្នុងការសរសេរកូដ កំហុស (Errors) គឺជារឿងដែលមិនអាចចៀសផុត។ វាអាចមកពី៖
*   កូដមានបញ្ហា (Syntax Error, Reference Error)
*   បញ្ហាបណ្តាញអុីនធឺណិត (Network Issue) ពេលកំពុងទាញយកទិន្នន័យ
*   អ្នកប្រើប្រាស់បញ្ចូលទិន្នន័យខុសទម្រង់

ប្រសិនបើមាន Error កើតឡើង ហើយយើងមិនបានរៀបចំទុកមុនទេ នោះ **JavaScript នឹងបញ្ឈប់ការដំណើរការកូដទាំងស្រុង (Crash)**។ ការធ្វើ **Error Handling** អនុញ្ញាតអោយយើងចាប់យកកំហុសនោះ ដោះស្រាយវាដោយសន្តិវិធី និងបន្តដំណើរការកម្មវិធីទៅមុខទៀត។

## 1. ការប្រើប្រាស់ `try...catch`

យន្តការសំខាន់បំផុតគឺ `try...catch`។
*   **`try`**: អ្នកដាក់កូដដែល "អាចនឹង" មានបញ្ហានៅទីនេះ។
*   **`catch`**: បើកូដក្នុង `try` មានបញ្ហាមែន វានឹងលោតចូលមក `catch` ភ្លាមៗដោយមិនធ្វើអោយកម្មវិធីគាំងឡើយ។

try {
  // ព្យាយាមដំណើរការកូដនេះ
  let message = "សួស្តី!";
  console.log(message);

  // ធ្វើអោយមានកំហុសដោយហៅ Function ដែលគ្មាននៅលើលោកនេះ!
  nonExistentFunction();

  // កូដបន្ទាប់ពី Error នឹងមិនដំណើរការទេ វានឹងលោតទៅ catch ភ្លាម
  console.log("អ្នកនឹងមិនឃើញសារមួយនេះទេ");

} catch (error) {
  // ចាប់យកកំហុសមកបង្ហាញ
  console.log("មានបញ្ហាកើតឡើង! នេះជាព័ត៌មានលម្អិត៖", error.message);
}

// កម្មវិធីនៅតែបន្តដំណើរការជាធម្មតា ដោយមិន Crash ទេ
console.log("កម្មវិធីនៅដើរធម្មតា!");
```

## 2. ការប្រើប្រាស់ `finally`

ជួនកាល អ្នកចង់អោយកូដមួយចំនួនដំណើរការដាច់ខាត ទោះជាមាន ឬមិនមាន Error ក៏ដោយ (ឧទាហរណ៍៖ ការលាក់ Loading Spinner ក្រោយពេលទាញយកទិន្នន័យចប់)។ នោះគឺជាតួនាទីរបស់ `finally`។

js
let isLoading = true;

try {
  console.log("កំពុងទាញយកទិន្នន័យ...");
  // ឧបមាថាមាន Error កើតឡើងទីនេះ
  throw new Error("ដាច់អុីនធឺណិត!");

} catch (error) {
  console.log("កំហុស៖", error.message);

} finally {
  // តែងតែដំណើរការជានិច្ច — លាក់ spinner
  isLoading = false;
  document.getElementById("spinner").style.display = "none";
  console.log("បញ្ចប់ប្រតិបត្តិការ។ Loading =", isLoading);
}
```

## 3. ការបង្កើតកំហុសដោយខ្លួនឯង (Throwing Custom Errors)

ពេលខ្លះ JavaScript មិនចាត់ទុកវាជា Error ទេ ប៉ុន្តែតាម Logic នៃកម្មវិធីរបស់អ្នក វាគឺជាកំហុស! អ្នកអាចប្រើពាក្យគន្លឹះ `throw` ដើម្បីបោះ Error នោះទៅក្នុង `catch` ដោយខ្លួនឯង។

function checkAge(age) {
  try {
    if (age < 0) {
      throw new Error("អាយុមិនអាចជាលេខអវិជ្ជមានបានទេ!");
    }
    if (age < 18) {
      console.log("អ្នកនៅក្មេងពេក។");
    } else {
      console.log("ស្វាគមន៍មកកាន់ប្រព័ន្ធ!");
    }
  } catch (error) {
    // ចាប់យក custom Error
    const msg = document.getElementById("ageError");
    msg.textContent = "❌ " + error.message;
    msg.className = "error";
  }
}

document.getElementById("ageForm").addEventListener("submit", (e) => {
  e.preventDefault();
  const age = Number(document.getElementById("ageInput").value);
  checkAge(age);
});

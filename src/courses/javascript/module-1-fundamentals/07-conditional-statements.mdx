## 🛤️ អ្វីទៅជា Conditional Statements?

ក្នុងជីវិតប្រចាំថ្ងៃ យើងតែងតែធ្វើការសម្រេចចិត្ត៖ *"បើមេឃភ្លៀង ខ្ញុំនឹងយកឆ័ត្រ បើមិនអញ្ចឹងទេ ខ្ញុំមិនយកទេ"*។ ក្នុងកូដក៏ដូចគ្នា **Conditional Statements** អនុញ្ញាតអោយកូដរបស់យើងដើរទៅតាមផ្លូវខុសៗគ្នា អាស្រ័យលើលក្ខខណ្ឌណាមួយ (ពិត ឬ មិនពិត)។

---

## 1. `if`, `else if`, និង `else`

នេះគឺជាទម្រង់ពេញនិយមបំផុតសម្រាប់ការធ្វើលក្ខខណ្ឌនៅក្នុង JavaScript។ យើងអាចប្រើប្រាស់វាទៅតាមកម្រិតស្មុគស្មាញនៃលក្ខខណ្ឌ៖

### ១.១. Simple `if` (លក្ខខណ្ឌមួយផ្លូវ)
ប្រើនៅពេលដែលយើងចង់ដំណើរការកូដតែក្នុងករណីដែលលក្ខខណ្ឌពិត (`true`) ប៉ុណ្ណោះ បើមិនពិតគឺមិនបាច់ធ្វើអ្វីឡើយ។

ជាដំបូង យើងប្រកាសអថេរ `hasMoney` សម្រាប់បញ្ជាក់ថាយើងមានលុយទិញកាហ្វេដែរឬទេ៖

```javascript
let hasMoney = true;
```

បន្ទាប់មក យើងសរសេរ `if` ធម្មតាដើម្បីពិនិត្យលក្ខខណ្ឌ៖

```javascript
if (hasMoney) {
  console.log("ទិញកាហ្វេញ៉ាំ! ☕");
}
```

---

### ១.២. `if...else` (លក្ខខណ្ឌពីរផ្លូវ)
ប្រើនៅពេលដែលយើងមានផ្លូវពីរជ្រើសរើស៖ បើលក្ខខណ្ឌពិត ធ្វើកិច្ចការមួយ បើមិនពិត ធ្វើកិច្ចការមួយទៀត។

ជាដំបូង យើងកំណត់ពិន្ទុសម្រាប់តេស្តសាកល្បង៖

```javascript
let score = 45;
```

បន្ទាប់មក យើងពិនិត្យមើលថាតើសិស្សប្រឡងជាប់ ឬធ្លាក់៖

```javascript
if (score >= 50) {
  console.log("ប្រឡងជាប់! 🎉");
} else {
  console.log("ប្រឡងធ្លាក់! ព្យាយាមម្តងទៀត។ 😢");
}
```

---

### ១.៣. `if...else if...else` (លក្ខខណ្ឌច្រើនផ្លូវ)
ប្រើនៅពេលដែលយើងមានលក្ខខណ្ឌត្រួតពិនិត្យច្រើនបន្តបន្ទាប់គ្នា (ចាប់ពី ៣ ផ្លូវឡើងទៅ)។

ជាដំបូង យើងប្រកាសអថេរ `time` ផ្ទុកម៉ោងសម្រាប់សាកល្បង៖

```javascript
let time = 14; // ម៉ោង 2 រសៀល (ប្រព័ន្ធ 24 ម៉ោង)
```

បន្ទាប់មក យើងសរសេរលក្ខខណ្ឌច្រើនជម្រើស ដើម្បីបង្ហាញសារស្វាគមន៍ទៅតាមម៉ោងនីមួយៗ៖

```javascript
if (time < 12) {
  console.log("អរុណសួស្តី!");
} else if (time < 18) {
  console.log("ទិវាសួស្តី!");
} else {
  console.log("សាយ័ន្តសួស្តី!");
}
```

អ្នកអាចមាន `else if` ច្រើនប៉ុណ្ណាក៏បាន ប៉ុន្តែ `else` (ចុងក្រោយគេបង្អស់) អាចមានតែមួយគត់ ឬមិនបាច់មានក៏បាន។

---

### ១.៤. Nested `if` (លក្ខខណ្ឌត្រួតគ្នាក្នុងគ្នា)
**Nested if** គឺជាការសរសេរ `if` មួយ នៅខាងក្នុងប្លុក `if` មួយផ្សេងទៀត។ វាត្រូវបានប្រើនៅពេលដែលកិច្ចការទី២ អាចកើតឡើងបាន លុះត្រាតែកិច្ចការទី១ ត្រូវបានសម្រេចជោគជ័យជាមុនសិន។

ជាដំបូង យើងបង្កើតអថេរពីរសម្រាប់ពិនិត្យ គឺការចូលគណនី (`isLoggedIn`) និងតួនាទីជាអ្នកគ្រប់គ្រង (`isAdmin`)៖

```javascript
let isLoggedIn = true;
let isAdmin = false;
```

បន្ទាប់មក យើងសរសេរលក្ខខណ្ឌត្រួតពិនិត្យ ២ ជាន់។ ជាន់ទី១ ពិនិត្យថាចូលគណនីហើយឬនៅ បើចូលហើយ ទើបពិនិត្យជាន់ទី២ ថាតើជា Admin ដែរឬទេ៖

```javascript
if (isLoggedIn) {
  console.log("ស្វាគមន៍មកកាន់ប្រព័ន្ធ!");
  
  if (isAdmin) {
    console.log("អ្នកមានសិទ្ធិចូលទៅកាន់ Control Panel។ ⚙️");
  } else {
    console.log("អ្នកមានសិទ្ធិត្រឹមជាអ្នកប្រើប្រាស់ធម្មតា។ 👤");
  }
} else {
  console.log("សូមចូលគណនីជាមុនសិន! 🔒");
}
```

## ⚖️ 2. Switch Statement

នៅពេលដែលអ្នកមាន `else if` ច្រើនពេក ដើម្បីប្រៀបធៀបជាមួយតម្លៃតែមួយ `switch` គឺជាជម្រើសដ៏ល្អ ព្រោះវាធ្វើអោយកូដងាយស្រួលអានជាង។

ជាដំបូង យើងប្រកាសអថេរ `day` ដើម្បីផ្ទុកឈ្មោះថ្ងៃសម្រាប់យកទៅពិនិត្យ៖

```javascript
let day = "Monday";
```

បន្ទាប់មក យើងប្រើប្រាស់បញ្ជា `switch` ដើម្បីពិនិត្យលើតម្លៃថ្ងៃ ជំនួសឱ្យការសរសេរ `if-else if` ច្រើនជាន់៖

```javascript
switch (day) {
  case "Monday":
    console.log("ថ្ងៃចន្ទ - ចាប់ផ្តើមធ្វើការវិញហើយ!");
    break; // បញ្ឈប់ការឆែកបន្តទៀត
  case "Friday":
    console.log("ថ្ងៃសុក្រ - ជិតដល់ចុងសប្តាហ៍ហើយ!");
    break;
  case "Saturday":
  case "Sunday":
    console.log("ថ្ងៃសម្រាក! ដើរលេង!"); // ដើរសម្រាប់ទាំង Saturday និង Sunday
    break;
  default:
    // ដើរប្រសិនបើគ្មាន case ណាមួយត្រូវសោះ (ស្រដៀងនឹង else)
    console.log("ជាថ្ងៃធ្វើការធម្មតា។");
}
```

> ⚠️ **កុំភ្លេច `break;`** នៅក្នុង `switch`! បើអ្នកភ្លេចដាក់វា កូដនឹងបន្តរត់ (Fall-through) ចូលទៅក្នុង case បន្ទាប់ទោះបីជាលក្ខខណ្ឌវាមិនត្រូវក៏ដោយ។

## ❓ 3. Ternary Operator

**Ternary Operator** គឺជាវិធីសរសេរកាត់របស់ `if...else` នៅក្នុងមួយបន្ទាត់! វាស័ក្តិសមបំផុតសម្រាប់លក្ខខណ្ឌសាមញ្ញ។

ជាដំបូង ឧបមាថាយើងមានលក្ខខណ្ឌសាមញ្ញមួយសម្រាប់ពិនិត្យអាយុដោយប្រើ `if-else` បែបធម្មតា៖

```javascript
let age = 20;
let statusNormal;

if (age >= 18) {
  statusNormal = "ពេញវ័យ";
} else {
  statusNormal = "អនីតិជន";
}
```

យើងអាចបង្រួមកូដ `if-else` ខាងលើឱ្យមកនៅត្រឹមតែមួយបន្ទាត់យ៉ាងខ្លី និងឆាប់រហ័សដោយប្រើប្រាស់ Ternary Operator (`? :`)៖

```javascript
// ប្រើប្រាស់ Ternary Operator ក្នុងបន្ទាត់តែមួយ
let statusTernary = age >= 18 ? "ពេញវ័យ" : "អនីតិជន";

console.log(statusTernary); // "ពេញវ័យ"
```

អ្នកអាចអានវាថា៖ *"តើ age ធំជាង ឬស្មើ 18 មែនទេ (`?`)? បើមែន យក `ពេញវ័យ`។ បើមិនមែន (`:`) យក `អនីតិជន`"*។

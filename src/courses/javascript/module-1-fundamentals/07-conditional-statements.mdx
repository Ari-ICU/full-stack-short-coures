## 🛤️ អ្វីទៅជា Conditional Statements?

ក្នុងជីវិតប្រចាំថ្ងៃ យើងតែងតែធ្វើការសម្រេចចិត្ត៖ *"បើមេឃភ្លៀង ខ្ញុំនឹងយកឆ័ត្រ បើមិនអញ្ចឹងទេ ខ្ញុំមិនយកទេ"*។ ក្នុងកូដក៏ដូចគ្នា **Conditional Statements** អនុញ្ញាតអោយកូដរបស់យើងដើរទៅតាមផ្លូវខុសៗគ្នា អាស្រ័យលើលក្ខខណ្ឌណាមួយ (ពិត ឬ មិនពិត)។

---

## 1. `if`, `else if`, និង `else`

នេះគឺជាទម្រង់ពេញនិយមបំផុតសម្រាប់ការធ្វើលក្ខខណ្ឌ។

ជាដំបូង យើងប្រកាសអថេរមួយឈ្មោះ `time` ដើម្បីរក្សាតម្លៃម៉ោងសម្រាប់សាកល្បង៖

```javascript
let time = 14; // ម៉ោង 2 រសៀល (ប្រព័ន្ធ 24 ម៉ោង)
```

បន្ទាប់មក យើងសរសេររចនាសម្ព័ន្ធលក្ខខណ្ឌ `if-else if-else` ដើម្បីត្រួតពិនិត្យ និងបង្ហាញសារស្វាគមន៍ទៅតាមតម្លៃម៉ោងជាក់ស្តែង៖

```javascript
if (time < 12) {
  // ដើរនៅពេលលក្ខខណ្ឌ (time < 12) គឺ true
  console.log("អរុណសួស្តី! ព្រឹកនេះស្រស់ស្រាយទេ?");
} else if (time < 18) {
  // ដើរនៅពេលលក្ខខណ្ឌខាងលើខុស ហើយលក្ខខណ្ឌនេះ true
  console.log("ទិវាសួស្តី! ញ៉ាំបាយថ្ងៃត្រង់ហើយនៅ?");
} else {
  // ដើរនៅពេលដែលលក្ខខណ្ឌខាងលើខុសទាំងអស់
  console.log("សាយ័ន្តសួស្តី!");
}
```

អ្នកអាចមាន `else if` ច្រើនប៉ុណ្ណាក៏បាន ប៉ុន្តែ `else` (ចុងក្រោយគេបង្អស់) អាចមានតែមួយគត់ ឬមិនបាច់មានក៏បាន។

## ⚖️ 2. Switch Statement

នៅពេលដែលអ្នកមាន `else if` ច្រើនពេក ដើម្បីប្រៀបធៀបជាមួយតម្លៃតែមួយ `switch` គឺជាជម្រើសដ៏ល្អ ព្រោះវាធ្វើអោយកូដងាយស្រួលអានជាង។

ជាដំបូង យើងប្រកាសអថេរ `day` ដើម្បីផ្ទុកឈ្មោះថ្ងៃសម្រាប់យកទៅពិនិត្យ៖

```javascript
let day = "Monday";
```

បន្ទាប់មក យើងប្រើប្រាស់បញ្ជា `switch` ដើម្បីពិនិត្យលើតម្លៃថ្ងៃ ជំនួសឱ្យការសរសេរ `if-else if` ច្រើនជាន់៖

```javascript
switch (day) {
  case "Monday":
    console.log("ថ្ងៃចន្ទ - ចាប់ផ្តើមធ្វើការវិញហើយ!");
    break; // បញ្ឈប់ការឆែកបន្តទៀត
  case "Friday":
    console.log("ថ្ងៃសុក្រ - ជិតដល់ចុងសប្តាហ៍ហើយ!");
    break;
  case "Saturday":
  case "Sunday":
    console.log("ថ្ងៃសម្រាក! ដើរលេង!"); // ដើរសម្រាប់ទាំង Saturday និង Sunday
    break;
  default:
    // ដើរប្រសិនបើគ្មាន case ណាមួយត្រូវសោះ (ស្រដៀងនឹង else)
    console.log("ជាថ្ងៃធ្វើការធម្មតា។");
}
```

> ⚠️ **កុំភ្លេច `break;`** នៅក្នុង `switch`! បើអ្នកភ្លេចដាក់វា កូដនឹងបន្តរត់ (Fall-through) ចូលទៅក្នុង case បន្ទាប់ទោះបីជាលក្ខខណ្ឌវាមិនត្រូវក៏ដោយ។

## ❓ 3. Ternary Operator

**Ternary Operator** គឺជាវិធីសរសេរកាត់របស់ `if...else` នៅក្នុងមួយបន្ទាត់! វាស័ក្តិសមបំផុតសម្រាប់លក្ខខណ្ឌសាមញ្ញ។

ជាដំបូង ឧបមាថាយើងមានលក្ខខណ្ឌសាមញ្ញមួយសម្រាប់ពិនិត្យអាយុដោយប្រើ `if-else` បែបធម្មតា៖

```javascript
let age = 20;
let statusNormal;

if (age >= 18) {
  statusNormal = "ពេញវ័យ";
} else {
  statusNormal = "អនីតិជន";
}
```

យើងអាចបង្រួមកូដ `if-else` ខាងលើឱ្យមកនៅត្រឹមតែមួយបន្ទាត់យ៉ាងខ្លី និងឆាប់រហ័សដោយប្រើប្រាស់ Ternary Operator (`? :`)៖

```javascript
// ប្រើប្រាស់ Ternary Operator ក្នុងបន្ទាត់តែមួយ
let statusTernary = age >= 18 ? "ពេញវ័យ" : "អនីតិជន";

console.log(statusTernary); // "ពេញវ័យ"
```

អ្នកអាចអានវាថា៖ *"តើ age ធំជាង ឬស្មើ 18 មែនទេ (`?`)? បើមែន យក `ពេញវ័យ`។ បើមិនមែន (`:`) យក `អនីតិជន`"*។

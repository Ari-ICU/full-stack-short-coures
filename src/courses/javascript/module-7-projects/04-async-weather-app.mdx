## ⛅ គម្រោងទី៤៖ Weather App Dashboard

គម្រោងចុងក្រោយនេះ គឺជាការសរុបមកវិញនូវរាល់មេរៀនទាំងអស់ដែលអ្នកបានរៀន រួមមាន **DOM Manipulation**, **Events**, **Async/Await**, និង **API Integration**។ យើងនឹងបង្កើតកម្មវិធីស្វែងរកអាកាសធាតុនៃទីក្រុងនានា។

## 2. កូដ JavaScript និងការហៅ API (Fetch & Render)

យើងនឹងប្រើប្រាស់ API សាកល្បង ឬ API ឥតគិតថ្លៃដើម្បីទាញយកទិន្នន័យអាកាសធាតុ។

```javascript
const cityInput = document.getElementById("city-input");
const searchBtn = document.getElementById("search-btn");
const loader = document.getElementById("loader");
const weatherInfo = document.getElementById("weather-info");
const cityName = document.getElementById("city-name");
const temperature = document.getElementById("temperature");
const condition = document.getElementById("condition");

// Function ទាញយកទិន្នន័យអាកាសធាតុ (Async/Await)
async function fetchWeather(city) {
  // បង្ហាញ Loader និងលាក់ព័ត៌មានអាកាសធាតុចាស់
  loader.classList.remove("hidden");
  weatherInfo.classList.add("hidden");
  
  // ប្រើប្រាស់ API Mock ឬ Free API
  const apiKey = "YOUR_API_KEY"; // បញ្ចូល API Key របស់អ្នក
  const url = `https://api.openweathermap.org/data/2.5/weather?q=${city}&units=metric&appid=${apiKey}`;
  
  try {
    const response = await fetch(url);
    
    if (!response.ok) {
      throw new Error("❌ រកមិនឃើញទីក្រុងនេះឡើយ!");
    }
    
    const data = await response.json();
    displayWeather(data); // បង្ហាញព័ត៌មានលើ screen
  } catch (error) {
    alert(error.message);
  } finally {
    // លាក់ Loader មិនថាជោគជ័យ ឬបរាជ័យ
    loader.classList.add("hidden");
  }
}

// Function បង្ហាញទិន្នន័យលើ DOM
function displayWeather(data) {
  cityName.textContent = `${data.name}, ${data.sys.country}`;
  temperature.textContent = `${Math.round(data.main.temp)}°C`;
  condition.textContent = data.weather[0].description;
  
  // បង្ហាញប្រអប់ព័ត៌មានអាកាសធាតុ
  weatherInfo.classList.remove("hidden");
}

// ភ្ជាប់ Event Listener ទៅលើប៊ូតុងស្វែងរក
searchBtn.addEventListener("click", () => {
  const city = cityInput.value.trim();
  if (city) {
    fetchWeather(city);
  } else {
    alert("សូមបញ្ចូលឈ្មោះទីក្រុង!");
  }
});
```

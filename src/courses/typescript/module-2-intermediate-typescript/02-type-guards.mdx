## 🛡️ អ្វីទៅជា Type Guards?

នៅក្នុងមេរៀនមុន អ្នកបានរៀនពីការប្រើប្រាស់ `typeof` ឬ `in` ដើម្បីឆែកលក្ខខណ្ឌ។ នេះគឺជា Type Guards ដែលមានស្រាប់។
ប៉ុន្តែពេលខ្លះ ការឆែកលក្ខខណ្ឌមានភាពស្មុគស្មាញ (ឧទាហរណ៍៖ ការឆែកទិន្នន័យ `User` ដែលបានមកពី API ថាមាន `name`, `email`, និង `id` ត្រឹមត្រូវឫអត់)។ 
ការសរសេរ `if` វែងៗរាល់ពេល គឺជារឿងហត់នឿយណាស់!

ដំណោះស្រាយគឺយើងអាចបង្កើត **អនុគមន៍ផ្ទាល់ខ្លួន (Custom Type Guard Function)** សម្រាប់ធ្វើរឿងនេះ។

## ការបង្កើត Type Guard (ប្រើប្រាស់ `is`)

ដើម្បីអោយអនុគមន៍របស់អ្នកក្លាយជា Type Guard អ្នកត្រូវប្រើប្រាស់អ្វីដែលគេហៅថា **Type Predicate** (`អថេរ is ប្រភេទ`) ជំនួសអោយ `boolean` នៅត្រង់ Return Type។

```ts
interface User {
  id: number;
  name: string;
}

// នេះគឺជា Custom Type Guard
// បើ return true នោះ 'value' គឺជា 'User' ប្រាកដមែន
function isUser(value: unknown): value is User {
  return (
    typeof value === "object" && // ត្រូវតែជា object
    value !== null &&            // មិនមែនជា null
    "id" in value &&             // ត្រូវមាន id
    "name" in value              // ត្រូវមាន name
  );
}
```

### របៀបយកមកប្រើប្រាស់

នៅពេលដែលអ្នកហៅប្រើអនុគមន៍ `isUser` នៅក្នុង `if` លក្ខខណ្ឌ TypeScript នឹងជឿជាក់លើវា ហើយបង្រួមប្រភេទ (Narrow) អោយអ្នកភ្លាមៗ!

```ts
function handleData(apiResponse: unknown) {
  // មុនពេលឆែក apiResponse មិនអាចប្រើបានទេ ព្រោះវា unknown
  
  if (isUser(apiResponse)) {
    // ចូលដល់ក្នុងនេះ apiResponse ត្រូវបានស្គាល់ថាជា User ពេញសិទ្ធិ!
    console.log("Welcome,", apiResponse.name);
  } else {
    console.log("ទិន្នន័យមិនត្រឹមត្រូវ!");
  }
}
```

ការសរសេរ Custom Type Guards គឺជារឿងដែលអ្នកអភិវឌ្ឍន៍ Senior តែងតែធ្វើនៅពេលពួកគេទាញយកទិន្នន័យពី Server មកបង្ហាញនៅលើ Frontend (ឧទាហរណ៍ ប្រើជាមួយ `fetch` ឬ `axios`)។

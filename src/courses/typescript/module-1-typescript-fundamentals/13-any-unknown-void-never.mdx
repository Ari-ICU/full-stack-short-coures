## 🎭 ប្រភេទពិសេសៗក្នុង TypeScript

ក្រៅពី `string` និង `number` TypeScript មានប្រភេទពិសេសៗចំនួន ៤ ទៀតដែលអ្នកគួរតែស្គាល់។

---

## 1. គ្រោះថ្នាក់នៃ `any` (អីក៏បាន)

នៅពេលអ្នកកំណត់ប្រភេទជា `any` វាមានន័យថាអ្នកកំពុងប្រាប់ TypeScript ថា៖ *"កុំខ្វល់ពីអថេរនេះអី ទុកអោយខ្ញុំអ្នកទទួលខុសត្រូវខ្លួនឯង!"*។ វាបិទមុខងារចាប់កំហុសទាំងអស់!

```ts
let obj: any = { x: 0 };

// TypeScript មិនចាប់កំហុសទេ ទោះបីជាគ្មានមុខងារ foo ក៏ដោយ! (តែវានឹង Error ពេល Run)
obj.foo(); 
obj = "hello"; // ប្តូរពី Object ទៅជាអក្សរក៏បាន
```

> ⚠️ **ប្រយ័ត្ន:** កុំប្រើ `any` អោយសោះប្រសិនបើអ្នកអាចជៀសវាងបាន ព្រោះវាធ្វើអោយការប្រឹងប្រែងសរសេរ TypeScript របស់អ្នកទៅជាអសារបង់! ការសរសេរ `any` ច្រើនពេក គេហៅថា "AnyScript" ។

---

## 2. ភាពមានសុវត្ថិភាពនៃ `unknown`

ស្រដៀងនឹង `any` ដែរ ប្រភេទ `unknown` អាចផ្ទុកតម្លៃអ្វីក៏បាន (លេខ អក្សរ Array)។ ប៉ុន្តែវាមានសុវត្ថិភាពជាង ត្រង់ថា **វាមិនអនុញ្ញាតអោយអ្នកយកវាទៅធ្វើប្រតិបត្តិការទេ រហូតទាល់តែអ្នកបញ្ជាក់ប្រាប់វាសិនថាវាជាអ្វីពិតប្រាកដ!**

```ts
let secretValue: unknown = "សួស្តី";

// ❌ Error! TypeScript មិនអោយហៅ toUpperCase ទេ ព្រោះវាមិនទាន់ប្រាកដថា secretValue គឺជាអក្សរពិតឫអត់។
// secretValue.toUpperCase(); 

// ✅ ត្រូវ! យើងត្រូវឆែកវាសិន គេហៅថា Type Narrowing
if (typeof secretValue === "string") {
  console.log(secretValue.toUpperCase()); // ឥឡូវ TypeScript អនុញ្ញាតហើយ!
}
```

## 3. `void` (គ្មានតម្លៃត្រលប់មកវិញ)

អ្នកនឹងឃើញ `void` ជាញឹកញាប់នៅក្នុងការកំណត់មុខងារ (Functions)។ វាមានន័យថា អនុគមន៍នេះ **មិនមាន return អ្វីត្រលប់មកវិញទេ** (គ្មាន return statement ឫគ្រាន់តែ return ទទេ)។ វាគ្រាន់តែដំណើរការកូដខាងក្នុងរួចក៏បញ្ចប់។

```ts
// អនុគមន៍នេះគ្រាន់តែ print ចោល មិនមានប្រគល់តម្លៃអ្វីអោយយើងទេ
function logMessage(msg: string): void {
  console.log("សាររបស់អ្នកគឺ:", msg);
  // គ្មានពាក្យ return ទេ
}
```

## 4. `never` (មិនអាចកើតឡើងជារៀងរហូត)

`never` គឺកម្រនឹងឃើញខ្លាំងណាស់។ វាមានន័យថា ជាមុខងារដែល **គ្មានថ្ងៃអាចរត់ចប់ (Complete) ឡើយ**។ 
ឧទាហរណ៍ អនុគមន៍ដែលតែងតែបោះកំហុស (Throw Error) ឫ អនុគមន៍ដែលមានរង្វិលជុំមិនចេះចប់ (Infinite loop)។

```ts
function crashApp(message: string): never {
  throw new Error(message); // កូដឈប់រត់ត្រឹមនេះ វាគ្មានថ្ងៃអាចរត់ដល់ចុងបញ្ចប់នៃអនុគមន៍ឡើយ
}
```

## 🧠 ការចងចាំរបស់ Component (`useState`)

ប្រសិនបើ **Props** គឺជាទិន្នន័យដែលគេហុចពីខាងក្រៅមកអោយ, **State** គឺជាទិន្នន័យ (ការចងចាំ) ផ្ទាល់ខ្លួនរបស់ Component នោះ។ រាល់ពេលដែល State ផ្លាស់ប្តូរ React នឹងគូរ Component នោះឡើងវិញ (Re-render) ដើម្បីបង្ហាញទិន្នន័យថ្មីនៅលើអេក្រង់។

យើងប្រើប្រាស់ **Hook** មួយឈ្មោះថា `useState` ដើម្បីបង្កើត State។

```jsx
import { useState } from 'react';

export default function Counter() {
  // [តម្លៃបច្ចុប្បន្ន, អនុគមន៍សម្រាប់ដូរតម្លៃ] = useState(តម្លៃដើម)
  const [count, setCount] = useState(0);

  return (
    <div className="p-10 text-center">
      <h1 className="text-3xl font-bold mb-4">ពិន្ទុ: {count}</h1>
      <button 
        className="bg-blue-500 text-white px-4 py-2 rounded"
        onClick={() => setCount(count + 1)} // ដូរតម្លៃ count កើន ១
      >
        បូក ១
      </button>
    </div>
  );
}
```

## ⚡ ការធ្វើអន្តរកម្មជាមួយខាងក្រៅ (`useEffect`)

**Side Effects** គឺជាសកម្មភាពអ្វីក៏ដោយដែលកើតឡើងក្រៅពីការគូរ UI ធម្មតា ឧទាហរណ៍៖ ការទាញយកទិន្នន័យពី Server (Fetch API), ការស្តាប់ Event របស់ Browser (ឧ. Window resize), ឬការកំណត់ Timer (`setTimeout`)។ 

យើងប្រើ `useEffect` ដើម្បីគ្រប់គ្រង Side Effects ទាំងនេះ។

```jsx
import { useState, useEffect } from 'react';

export default function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // នេះជាកូដទាញយកទិន្នន័យពី Internet
    fetch('https://jsonplaceholder.typicode.com/users')
      .then(response => response.json())
      .then(data => {
        setUsers(data);
        setLoading(false);
      });
      
  }, []); // [] មានន័យថាទាញយកតែម្តងគត់ពេលបើកទំព័រដំបូង

  if (loading) return <p>កំពុងទាញយកទិន្នន័យ...</p>;

  return (
    <ul>
      {users.map(user => (
        <li key={user.id} className="py-2 border-b">{user.name}</li>
      ))}
    </ul>
  );
}
```

## 🌍 ទិន្នន័យសកលដោយមិនបាច់បញ្ជូនតៗគ្នា (`useContext`)

ស្រមៃថាអ្នកមាន Component ធំមួយ ហើយអ្នកត្រូវបញ្ជូនទិន្នន័យ (ឧ. ឈ្មោះអ្នកប្រើប្រាស់ដែលបាន Login) ទៅកាន់ Component តូចៗដែលនៅជ្រៅៗ (Child របស់ Child របស់ Child)។ ការបញ្ជូន Props តៗគ្នាបែបនេះហៅថា **Prop Drilling** ដែលធ្វើអោយកូដរញ៉េរញ៉ៃ។

**Context API (`useContext`)** អនុញ្ញាតអោយយើងបង្កើត "ឃ្លាំងទិន្នន័យសកល" ដើម្បីអោយ Component ណាក៏អាចទាញយកមកប្រើបានដែរ ដោយមិនចាំបាច់បញ្ជូនតៗគ្នា។

**របៀបបង្កើត និងប្រើប្រាស់ Context:**

```jsx
import { createContext, useContext, useState } from 'react';

// ១. បង្កើត Context
const ThemeContext = createContext();

// ២. Component តូច (Child) ដែលត្រូវការទិន្នន័យនោះ
function Display() {
  const theme = useContext(ThemeContext); // ទាញយកទិន្នន័យពី Context មកប្រើ
  return <p>Theme បច្ចុប្បន្នគឺ: {theme}</p>;
}

// ៣. Component ធំ (Parent) ដែលជាអ្នកផ្តល់ទិន្នន័យ (Provider)
export default function App() {
  const [theme, setTheme] = useState('light');

  return (
    // រុំកូនៗជាមួយនឹង Provider ដើម្បីបញ្ជូនទិន្នន័យ
    <ThemeContext.Provider value={theme}>
      <div className={`p-10 ${theme === 'dark' ? 'bg-black text-white' : 'bg-white text-black'}`}>
        <button onClick={() => setTheme(theme === 'light' ? 'dark' : 'light')}>
          ប្តូរ Theme
        </button>
        
        {/* ចំណាំ: Display មិនបានទទួល props ពីខាងលើផ្ទាល់ទេ វាទាញពី Context */}
        <Display /> 
      </div>
    </ThemeContext.Provider>
  );
}
```

**សន្និដ្ឋាន:** `useState` ធ្វើអោយ Component មានជីវិតដោយសារការចងចាំ, `useEffect` អនុញ្ញាតអោយវាទាក់ទងជាមួយពិភពខាងក្រៅ, ហើយ `useContext` ជួយសម្រួលការបញ្ជូនទិន្នន័យ។ ប៉ុន្តែបើទិន្នន័យសកលរបស់អ្នកមានទំហំធំ និងស្មុគស្មាញ (ឧ. កន្ត្រកទំនិញ) អ្នកនឹងត្រូវការឧបករណ៍ដែលខ្លាំងជាងនេះ គឺ **Redux Toolkit** ដែលយើងនឹងរៀននៅមេរៀនបន្ទាប់!

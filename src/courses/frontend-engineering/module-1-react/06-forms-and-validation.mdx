## 📝 ការគ្រប់គ្រង Form តាមបែបទំនើប

ការបង្កើត Form នៅក្នុង React ពីមុន គឺជារឿងដែលគួរអោយធុញទ្រាន់ ព្រោះអ្នកត្រូវបង្កើត `useState` សម្រាប់ Input នីមួយៗ ហើយសរសេរ Function `onChange` ដោយខ្លួនឯង ដែលធ្វើអោយកូដវែង និងដើរយឺតនៅពេលមាន Input ច្រើន។

ដើម្បីដោះស្រាយបញ្ហានេះ ពិភពលោកបច្ចុប្បន្ននិយមប្រើប្រាស់ **React Hook Form** ផ្គូផ្គងជាមួយ **Zod**។
*(ដំឡើង: `npm install react-hook-form zod @hookform/resolvers`)*

- **React Hook Form:** ជួយគ្រប់គ្រងទិន្នន័យ Form ដោយមិនចាំបាច់មានការ Re-render ញឹកញាប់។
- **Zod:** ជាអ្នកបង្កើតច្បាប់ (Schema) ដើម្បីកំណត់ថា ទិន្នន័យបែបណាទើបអាចទទួលយកបាន (ឧ. អ៊ីមែលត្រូវតែមានសញ្ញា @ ឫ លេខសម្ងាត់ត្រូវតែលើសពី ៨ តួអក្សរ)។

## 🛠️ ការប្រើប្រាស់ React Hook Form និង Zod

ខាងក្រោមនេះជាការបង្កើត Form ចុះឈ្មោះ (Registration Form) ពេញលេញមួយ៖

### ១. ការបង្កើត Zod Schema (ច្បាប់ទិន្នន័យ)

ជាដំបូង យើងត្រូវប្រាប់ Zod ពីទម្រង់ដែលយើងចង់បាន ព្រមទាំងសារព្រមាន (Error messages)។

```javascript
import { z } from "zod";

const registerSchema = z.object({
  username: z.string().min(3, "ឈ្មោះត្រូវមានយ៉ាងហោចណាស់ ៣ តួអក្សរ"),
  email: z.string().email("ទម្រង់អ៊ីមែលមិនត្រឹមត្រូវទេ"),
  password: z.string().min(8, "លេខសម្ងាត់ត្រូវមានយ៉ាងហោចណាស់ ៨ តួអក្សរ")
});
```

### ២. ការភ្ជាប់ទៅកាន់ Component

បន្ទាប់មក យើងប្រើ `useForm` ពី React Hook Form ហើយភ្ជាប់វាជាមួយ Zod តាមរយៈ `zodResolver`។

```jsx
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { z } from "zod";

// ១. Schema
const registerSchema = z.object({
  username: z.string().min(3, "ឈ្មោះត្រូវមានយ៉ាងហោចណាស់ ៣ តួអក្សរ"),
  email: z.string().email("ទម្រង់អ៊ីមែលមិនត្រឹមត្រូវទេ"),
  password: z.string().min(8, "លេខសម្ងាត់ត្រូវមានយ៉ាងហោចណាស់ ៨ តួអក្សរ")
});

export default function RegisterForm() {
  
  // ២. រៀបចំ Form
  const { 
    register,     // ប្រើសម្រាប់ភ្ជាប់ Input ទៅកាន់ Form
    handleSubmit, // ប្រើពេល Submit
    formState: { errors } // ផ្ទុកសារព្រមាននៅពេល Validation មិនឆ្លងកាត់
  } = useForm({
    resolver: zodResolver(registerSchema)
  });

  // ៣. Function ដែលដំណើរការពេល Submit "ជោគជ័យ" គ្មាន Error
  const onSubmit = (data) => {
    console.log("ទិន្នន័យត្រឹមត្រូវ:", data);
    // នៅទីនេះ អ្នកអាចបញ្ជូនទិន្នន័យ (data) ទៅកាន់ API បាន
  };

  return (
    <div className="p-10 max-w-md mx-auto">
      <h2 className="text-2xl font-bold mb-4">បង្កើតគណនី</h2>
      
      <form onSubmit={handleSubmit(onSubmit)} className="flex flex-col gap-4">
        
        {/* វាលបញ្ជូលឈ្មោះ */}
        <div>
          <label className="block mb-1">ឈ្មោះអ្នកប្រើប្រាស់</label>
          <input 
            {...register("username")} // ភ្ជាប់ Input!
            className="border p-2 w-full rounded" 
            placeholder="ឧ. ratha"
          />
          {/* បង្ហាញ Error បើមាន */}
          {errors.username && <p className="text-red-500 text-sm mt-1">{errors.username.message}</p>}
        </div>

        {/* វាលបញ្ជូលអ៊ីមែល */}
        <div>
          <label className="block mb-1">អ៊ីមែល</label>
          <input 
            {...register("email")} 
            className="border p-2 w-full rounded" 
            placeholder="email@example.com"
          />
          {errors.email && <p className="text-red-500 text-sm mt-1">{errors.email.message}</p>}
        </div>

        {/* វាលបញ្ជូលលេខសម្ងាត់ */}
        <div>
          <label className="block mb-1">លេខសម្ងាត់</label>
          <input 
            type="password"
            {...register("password")} 
            className="border p-2 w-full rounded" 
          />
          {errors.password && <p className="text-red-500 text-sm mt-1">{errors.password.message}</p>}
        </div>

        {/* ប៊ូតុង Submit */}
        <button type="submit" className="bg-blue-600 text-white p-2 rounded hover:bg-blue-700">
          ចុះឈ្មោះ
        </button>

      </form>
    </div>
  );
}
```
